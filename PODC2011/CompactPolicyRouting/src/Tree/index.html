import Papers.PODC2011.CompactPolicyRouting.Paths import Cslib.Foundations.Combinatorics.SimpleGraph.GreedySpanner /-! # Mapping to a tree (Lemma 4.10) An algebra `A` *maps to a tree* if every connected weighted graph has a spanning tree whose (unique) tree paths are all preferred paths. Lemma 4.10: monotone + selective algebras map to a tree, and conversely (the paper assumes delimitedness for the converse; we show it is not needed). * Sufficiency (`exists_preferred_spanningTree`, `lemma_4_10_mapsToTree`): with `⊕ = max` a minimum spanning tree works. Instead of formalizing Kruskal we reuse CSLib's greedy spanner with a huge stretch (which *is* Kruskal's algorithm), run on integer ranks of the weights: it outputs a spanning tree with the cycle property (every graph edge `uv` is bypassed by a tree walk of no heavier edges). Every tree path is then dominated edgewise by any other path between its endpoints. * Necessity (`lemma_4_10_converse`): on the triangle, every spanning tree misses one edge, and three choices of edge weights (Fig. 4.1) refute non-monotone and non-selective algebras. -/ namespace Podc.CompactPolicyRouting open SimpleGraph RoutingAlgebra open scoped NNReal universe u variable {W : Type u} [LinearOrder W] /-- `A` maps to a tree: every connected (finite) weighted graph has a spanning tree in which, for all `s, t`, the unique `s – t` tree path is a preferred `s – t` path of the graph. -/ def RoutingAlgebra.MapsToTree (A : RoutingAlgebra W) : Prop := ∀ (V : Type) [Finite V] (G : SimpleGraph V), G.Connected → ∀ wt : Sym2 V → W, ∃ T : SimpleGraph V, ∃ hT : T ≤ G, T.IsTree ∧ ∀ (s t : V) (p : T.Walk s t), p.IsPath → IsPreferred A wt (p.mapLe hT) variable {A : RoutingAlgebra W} /-! ### Composition as maximum -/ /-- In a monotone selective algebra, composition of extended weights is the maximum. -/ theorem mul_eq_max (hM : A.IsMonotone) (hS : A.IsSelective) (x y : A.Ext) : x * y = max x y := by rcases Ext.cases x with rfl | ⟨a, rfl⟩ · rw [one_mul, max_eq_right (one_le y)] rcases Ext.cases y with rfl | ⟨b, rfl⟩ · rw [mul_one, max_eq_left (one_le _)] · rw [emb_mul_emb, op_eq_max hM hS] exact Monotone.map_max (fun _ _ h => emb_le_emb.2 h) theorem list_prod_le (hM : A.IsMonotone) (hS : A.IsSelective) {l : List A.Ext} {c : A.Ext} (h : ∀ x ∈ l, x ≤ c) : l.prod ≤ c := by induction l with | nil => exact one_le c | cons x l ih => rw [List.prod_cons, mul_eq_max hM hS] exact max_le (h x (by simp)) (ih fun y hy => h y (by simp [hy])) theorem le_list_prod (hM : A.IsMonotone) {l : List A.Ext} {x : A.Ext} (hx : x ∈ l) : x ≤ l.prod := by induction l with | nil => simp at hx | cons y l ih => rw [List.prod_cons] rcases List.mem_cons.1 hx with rfl | hx · exact hM.le_mul' _ _ · exact (ih hx).trans (hM.le_mul _ _) /-- With `⊕ = max`: if every edge of `p` is not heavier than some edge of `q`, then `w(p) ⪯ w(q)`. -/ theorem weight_le_of_forall_exists (hM : A.IsMonotone) (hS : A.IsSelective) {V : Type*} {wt : Sym2 V → W} {G H : SimpleGraph V} {s t s' t' : V} (p : G.Walk s t) (q : H.Walk s' t') (h : ∀ e ∈ p.edges, ∃ f ∈ q.edges, wt e ≤ wt f) : weight A wt p ≤ weight A wt q := by unfold weight refine list_prod_le hM hS fun x hx => ?_ obtain ⟨e, he, rfl⟩ := List.mem_map.1 hx obtain ⟨f, hf, hef⟩ := h e he exact (emb_le_emb.2 hef).trans (le_list_prod hM (List.mem_map.2 ⟨f, hf, rfl⟩)) /-! ### Minimum spanning trees with the cycle property -/ -- CSLib candidate /-- Every finite connected graph with edge weights in a linear order has a spanning tree with the cycle property: every edge `uv` of the graph is bypassed by a tree walk all of whose edges are not heavier than `uv`. (A Kruskal tree; obtained from CSLib's greedy spanner with stretch `2|V| - 1` on integer ranks of the weights.) -/ theorem exists_isTree_forall_walk {V : Type*} [Finite V] {G : SimpleGraph V} (hG : G.Connected) {α : Type*} [LinearOrder α] (wt : Sym2 V → α) : ∃ T ≤ G, T.IsTree ∧ ∀ u v, G.Adj u v → ∃ p : T.Walk u v, ∀ f ∈ p.edges, wt f ≤ wt s(u, v) := by classical have := Fintype.ofFinite V let S : Finset α := Finset.univ.image wt let r : Sym2 V → ℕ := fun e => (S.filter (· < wt e)).card have hr : ∀ e f, r f ≤ r e → wt f ≤ wt e := by intro e f hfe by_contra hlt rw [not_le] at hlt refine absurd hfe (not_le.2 (Finset.card_lt_card ?_)) refine (Finset.ssubset_iff_of_subset ?_).2 ⟨wt e, ?_, ?_⟩ · intro x hx simp only [Finset.mem_filter] at hx ⊢ exact ⟨hx.1, hx.2.trans hlt⟩ · simp [S, hlt] · simp let w : Sym2 V → ℝ≥0 := fun e => (r e : ℝ≥0) set n := Fintype.card V have hn : 1 ≤ n := Fintype.card_pos_iff.2 hG.nonempty obtain ⟨H, hH⟩ := exists_isGreedySpanner G w ((2 * n - 1 : ℕ) : ℝ≥0) have hsp := hH.isWSpanner (Nat.one_le_cast.2 (by omega)) obtain ⟨hHG, hwalk⟩ := hH.forall_walk have hcyc := hH.noShortCycles (k := n) le_rfl refine ⟨H, hHG, ⟨?_, ?_⟩, ?_⟩ · have := hG.nonempty exact Connected.mk fun u v => hsp.reachable (hG.preconnected u v) · intro v c hc have hlen := hcyc v c hc cases c with | nil => exact hc.ne_nil rfl | cons h p => have hp := ((Walk.cons_isCycle_iff p h).1 hc).1 have := hp.length_lt simp only [Walk.length_cons] at hlen omega · intro u v huv by_cases hH' : H.Adj u v · exact ⟨Walk.cons hH' Walk.nil, by simp⟩ · obtain ⟨p, hp⟩ := hwalk u v huv hH' exact ⟨p, fun f hf => hr _ _ (Nat.cast_le.1 (hp f hf))⟩ /-!
