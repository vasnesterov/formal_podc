import Papers.PODC2017.ZigzagCausality.Coordination /-! # Non-vacuity of Theorem 3: a flooding full-information protocol implementing `Late` The fork of Figure 1: `C = 0` sends to `A = 1` and `B = 2`, with `U_CA = 1` and `L_CB = U_CB = 2`, so `L_CB ≥ U_CA + 1`. All processes flood their full local state (an FFIP). `B` performs `b` when it receives from `C` a state in which `C` has just received `μ_go`. This implements `Late⟨a --1--> b⟩` (`implementsLate`), and in a concrete run `C` receives `μ_go` and `B` performs `b` (`performs_exists`), so the hypotheses of `thm3_late` are satisfiable. -/ set_option linter.unusedSectionVars false namespace Podc.ZigzagCausality namespace NonVacuity open Sys /-- Channels `0 → 1`, `0 → 2`. -/ def ch : Fin 3 → Fin 3 → Bool | 0, 1 => true | 0, 2 => true | _, _ => false /-- Bounds: `L₀₁ = U₀₁ = 1`, `L₀₂ = U₀₂ = 2`. -/ def lu : Fin 3 → Fin 3 → ℕ | 0, 2 => 2 | _, _ => 1 def net : Net (Fin 3) where Chan i j := ch i j = true L := lu U := lu one_le_L i j _ := by fin_cases i <;> fin_cases j <;> decide L_le_U _ _ _ := le_rfl abbrev LS := LState (Fin 3) Unit ℕ Unit abbrev BN := BNode (Fin 3) Unit ℕ Unit instance : Countable LS := by refine Function.Injective.countable (f := fun ℓ : LS => (ℓ.init, ℓ.hist.map fun b => (b.msgs, b.ext))) ?_ rintro ⟨a, h⟩ ⟨a', h'⟩ he simp only [Prod.mk.injEq] at he obtain ⟨-, he⟩ := he have ha : a = a' := Subsingleton.elim _ _ subst ha have : h = h' := by refine List.map_injective_iff.2 ?_ he rintro ⟨m, e⟩ ⟨m', e'⟩ hh simp only [Prod.mk.injEq] at hh obtain ⟨rfl, rfl⟩ := hh rfl rw [this] noncomputable def enc : LS → ℕ := Classical.choose (Countable.exists_injective_nat LS) theorem enc_injective : Function.Injective enc := Classical.choose_spec (Countable.exists_injective_nat LS) /-- The flooding full-information protocol. -/ noncomputable def γ : Sys (Fin 3) Unit ℕ Unit where net := net send i ℓ j := if ch i j = true then some (enc ℓ) else none G0 := Set.univ theorem isFFIP : γ.IsFFIP := by refine ⟨?_, ?_⟩ · intro i j j' ℓ ℓ' m h h' simp only [γ] at h h' split_ifs at h h' cases h; exact enc_injective (Option.some.inj h'.symm) · intro i ℓ j hc simp only [γ] rw [if_pos (show ch i j = true from hc)]; rfl /-- `C` has just received `μ_go`. -/ def goState (ℓ : LS) : Prop := ∃ b, ℓ.hist.getLast? = some b ∧ () ∈ b.ext /-- `B` performs `b` when its last batch contains a message from `C` whose content is a state of `C` that has just received `μ_go`. -/ def doB (ℓ : LS) : Prop := ∃ b, ℓ.hist.getLast?
