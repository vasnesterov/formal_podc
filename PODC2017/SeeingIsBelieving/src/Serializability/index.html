= some (H.ops T)[m] := List.getElem?_eq_getElem hm have hmem : (H.ops T)[m] ∈ H.ops T := List.getElem_mem hm generalize (H.ops T)[m] = o at hm' hmem cases o with | write k => exact (e.readState_write hm').2 le_rfl | read k v => refine (e.readState_read hm').2 ⟨le_rfl, ?_⟩ by_cases hint : v = some T ∧ Op.write k ∈ (H.ops T).take m · exact Or.inr hint left refine e.state_eq_of_no_next vo (hvo k) ?_ ?_ · rintro V rfl by_cases hVT : V = T · subst hVT; exact absurd ⟨rfl, hwf.read_own V hT m k hm'⟩ hint have hwV := hwf.read_other T hT k V hmem hVT have hVc : V ∈ H.committed := by by_contra hc; exact hG1 (Or.inl ⟨T, hT, k, V, hmem, hc⟩) exact ⟨hVc, hwV, hedge V T (Or.inl (Or.inr ⟨hVc, hT, hVT, k, hwV, hmem⟩))⟩ · intro N Z hN hNZ hZ hwZ hZT have hNZ' := le_of_reflTransGen e.idx (fun a b h => hedge a b (Or.inl (Or.inl h))) hNZ by_cases hNT : N = T · subst hNT; omega · have := hedge T N (Or.inr ⟨hT, Ne.symm hNT, k, v, hmem, hN⟩) omega end Podc.SeeingIsBelieving
