import Podc.Foundations.Combinatorics.SimpleGraph.EdgeColoring.Basic /-! # Round-scheduled greedy edge colouring Lemma 5.1 and Theorem 5.2 of the paper colour the edges of a graph round by round: some edges are already coloured (`pre`, colouring `c₀`); every other edge `e` is coloured in round `time e` by one of its endpoints, which picks a colour `< K` different from the colours of all adjacent edges that are pre-coloured or coloured in an earlier round **or in the same round** (`EdgeSchedule.Follows`). The schedules of the paper guarantee that two adjacent edges coloured in the same round are chosen by their common endpoint, so this rule can be executed locally (see `Lemma51`/`Theorem52`). * `EdgeSchedule.Follows.isEdgeColoring`: every colouring following the rule is proper (given that `c₀` is proper on the pre-coloured edges). * `EdgeSchedule.exists_follows`: the rule can always be followed if every edge has fewer than `K` adjacent edges that are pre-coloured or not scheduled later ("load" `< K`). -- CSLib candidate: `Cslib/Foundations/Combinatorics/SimpleGraph/EdgeColoring/Scheduled.lean` -/ open Finset namespace Podc.DeltaEdgeColoring set_option linter.unusedSectionVars false variable {V : Type*} [Fintype V] [DecidableEq V] /-- A colouring schedule: `pre e` means `e` is coloured beforehand; otherwise `e` is coloured in round `time e`. -/ structure EdgeSchedule (V : Type*) where /-- The pre-coloured edges. -/ pre : Sym2 V → Prop /-- The round in which a non-pre-coloured edge is coloured. -/ time : Sym2 V → ℕ namespace EdgeSchedule variable (σ : EdgeSchedule V) (G : SimpleGraph V) /-- The rule of the round-by-round algorithm: pre-coloured edges keep their colour `c₀`; every other edge `uv` gets a colour `< K` that differs from the colour of every other edge `uw` at `u` that is pre-coloured or coloured no later than `uv` (this is required at both endpoints, as `u` ranges over both). -/ def Follows (c₀ c : Sym2 V → ℕ) (K : ℕ) : Prop := (∀ ⦃u v⦄, G.Adj u v → σ.pre s(u, v) → c s(u, v) = c₀ s(u, v)) ∧ ∀ ⦃u v⦄, G.Adj u v → ¬ σ.pre s(u, v) → c s(u, v) < K ∧ ∀ ⦃w⦄, G.Adj u w → w ≠ v → (σ.pre s(u, w) ∨ σ.time s(u, w) ≤ σ.time s(u, v)) → c s(u, w) ≠ c s(u, v) variable {σ G} /-- Every colouring that follows the rule is a proper edge colouring, provided that the pre-colouring is proper on the pre-coloured edges. -/ theorem Follows.isEdgeColoring {c₀ c : Sym2 V → ℕ} {K : ℕ} (hf : σ.Follows G c₀ c K) (hc₀ : ∀ ⦃u v w⦄, G.Adj u v → G.Adj u w → σ.pre s(u, v) → σ.pre s(u, w) → c₀ s(u, v) = c₀ s(u, w) → v = w) : G.IsEdgeColoring c := by intro u v w huv huw he by_contra hvw by_cases hpv : σ.pre s(u, v) <;> by_cases hpw : σ.pre s(u, w) · rw [hf.1 huv hpv, hf.1 huw hpw] at he exact hvw (hc₀ huv huw hpv hpw he) · exact (hf.2 huw hpw).2 huv hvw (Or.inl hpv) he · exact (hf.2 huv hpv).2 huw (Ne.symm hvw) (Or.inl hpw) he.symm · rcases le_total (σ.time s(u, w)) (σ.time s(u, v)) with h | h · exact (hf.2 huv hpv).2 huw (Ne.symm hvw) (Or.inr h) he.symm · exact (hf.2 huw hpw).2 huv hvw (Or.inr h) he /-- A colouring following the rule uses the colours `< max k₀ K` if `c₀` uses colours `< k₀` on the pre-coloured edges. -/ theorem Follows.isProperEdgeColoring {c₀ c : Sym2 V → ℕ} {K k₀ : ℕ} (hf : σ.Follows G c₀ c K) (hc₀ : ∀ ⦃u v w⦄, G.Adj u v → G.Adj u w → σ.pre s(u, v) → σ.pre s(u, w) → c₀ s(u, v) = c₀ s(u, w) → v = w) (hk₀ : ∀ ⦃u v⦄, G.Adj u v → σ.pre s(u, v) → c₀ s(u, v) < k₀) : G.IsProperEdgeColoring c (max k₀ K) := by refine ⟨fun u v huv => ?_, hf.isEdgeColoring hc₀⟩ by_cases hp : σ.pre s(u, v) · rw [hf.1 huv hp]; exact lt_max_of_lt_left (hk₀ huv hp) · exact lt_max_of_lt_right (hf.2 huv hp).1 open Classical in /-- The load of the edge `uv` at its endpoint `u`: the other edges `uw` that are pre-coloured or not scheduled after `uv`. -/ noncomputable def load (σ : EdgeSchedule V) (G : SimpleGraph V) (u v : V) : Finset V := {w | G.Adj u w ∧ w ≠ v ∧ (σ.pre s(u, w) ∨ σ.time s(u, w) ≤ σ.time s(u, v))} open Classical in /-- **Feasibility of the rule.** If for every non-pre-coloured edge `uv` fewer than `K` adjacent edges are pre-coloured or not scheduled after `uv`, then some colouring follows the rule.
