import Papers.PODC2015.AlmostStableMatchings.Prefs /-! # The deterministic algorithm `ASM` (Section 3) Ostrovsky, Rosenbaum, *Fast Distributed Almost Stable Matchings*, PODC 2015, Algorithms 1–3. * The state of a man is `ManSt` = his remaining (not yet rejected) acceptable partners `Q`, his partner `p` and his set `A` of active potential mates. A woman stores her partner. (The woman's list `Q^w` of the paper is not stored: she sends `REJECT` to *all* acceptable men in a worse or equal quantile than her new partner, including men who have already removed her; this has no effect on any state.) * The maximal-matching subroutine of Theorem 2.7 is a black box `MMOracle`: any function returning, for every bipartite graph `G₀`, a maximal matching of `G₀` (Definition 2.6). * `prStep` is `ProposalRound` (Algorithm 1), assembled from the local rules `proposers` (step 1: `PROPOSE` messages received by a woman), `accepts` (step 2: `ACCEPT` messages), the subroutine (step 3), `rejects` (step 4: `REJECT` messages), `womanUpdate` and `manUpdate` (steps 4–5). Every node's new state depends only on its own state, the one-bit messages from its neighbours and its own partner in `M₀`. * `quantileMatch` is Algorithm 2, `outer` is one iteration of the outer loop of Algorithm 3 (the men with `|Q| ≥ 2^i` at the beginning of the iteration run `ℓ` calls of `quantileMatch`), and `asm` runs the iterations `i = 0, …, L`. -/ set_option linter.unusedSectionVars false (those with `w ∈ A`). -/ def proposers (s : St M W) (w : W) : Finset M := (P.NW w).filter fun m => w ∈ s.A m /-- Step 2: `w` sends `ACCEPT` to the proposers of her best quantile receiving proposals. ≠ m₀` in a lesser or equal quantile than `m₀`. -/ def rejects (w : W) (p0 : Option M) (m : M) : Bool := match p0 with | none => false | some m0 => decide (m ∈ P.NW w ∧ m ≠ m0 ∧ P.qW k w m0 ≤ P.qW k w m) /-- Step 4 (women): a woman matched in `M₀` takes her new partner. -/ def womanUpdate (pw p0 : Option M) : Option M := p0.or pw /-- Steps 4–5 (men): given his partner `p0` in `M₀` and the `REJECT` messages `rej`, a man removes the rejecting women from `Q` and `A`; if matched in `M₀` he takes the new partner and empties `A`; if his partner rejected him he becomes unmatched. -/ def manUpdate (σ : ManSt W) (p0 : Option W) (rej : W → Bool) : ManSt W where Q := σ.Q.filter fun w => rej w = false p := p0.or (σ.p.filter fun w => !rej w) A := if p0.isSome then ∅ else σ.A.filter fun w => rej w = false def mmOut (mm : MMOracle M W) (s : St M W) : BMatching M W := mm.run (P.acceptGraph k s) /-- `ProposalRound(Q, k, A)` (Algorithm 1). -/ def prStep (mm : MMOracle M W) (s : St M W) : St M W where man m := manUpdate (s.man m) ((P.mmOut k mm s).pm m) (fun w => P.rejects k w ((P.mmOut k mm s).pw w) m) pw w := womanUpdate (s.pw w) ((P.mmOut k mm s).pw w) /-! ## QuantileMatch (Algorithm 2) and ASM (Algorithm 3) -/ /-- The best nonempty quantile `Q_i` of a man's remaining list. -/ def bestQ (m : M) (Q : Finset W) : Finset W := Q.filter fun w => ∀ w' ∈ Q, P.qM k m w ≤ P.qM k m w' /-- First lines of `QuantileMatch`: every participating unmatched man sets `A ← Q_i`. -/ def initQM (act : Finset M) (s : St M W) : St M W where man m := if m ∈ act ∧ s.pm m = none then { s.man m with A := P.bestQ k m (s.Q m) } else s.man m pw := s.pw /-- `QuantileMatch(Q, k)` run by the men in `act` (Algorithm 2): `k` calls of `ProposalRound`. -/ def quantileMatch (mm : MMOracle M W) (act : Finset M) (s : St M W) : St M W := (P.prStep k mm)^[k] (P.initQM k act s) /-- The men active in iteration `i` of the outer loop: those with `|Q| ≥ 2^i`. -/ def active (_P : Prefs M W) (i : ℕ) (s : St M W) : Finset M := Finset.univ.filter fun m => 2 ^ i ≤ (s.Q m).card /-- Iteration `i` of the outer loop of `ASM`: the active men run `ℓ` calls of `QuantileMatch`. -/ def outer (mm : MMOracle M W) (ℓ i : ℕ) (s : St M W) : St M W := (P.quantileMatch k mm (P.active i s))^[ℓ] s /-- Initial state: full lists, nobody matched, `A = ∅`. -/ def init : St M W where man m := ⟨P.NM m, none, ∅⟩ pw _ := none /-- State after the outer iterations `0, …, i - 1`. -/ def stage (mm : MMOracle M W) (ℓ : ℕ) : ℕ → St M W | 0 => P.init | i + 1 => P.outer k mm ℓ i (stage mm ℓ i) /-- `ASM` with `k` quantiles, `ℓ` calls of `QuantileMatch` per outer iteration and outer iterations `i = 0, …, L`. -/ def asm (mm : MMOracle M W) (ℓ L : ℕ) : St M W := P.stage k mm ℓ (L + 1) /-! ## The paper's parameters -/ /-- `k = ⌈8/ε⌉`. -/ noncomputable def kOf (ε : ℝ) : ℕ := ⌈8 / ε⌉₊ /-- `ℓ = ⌈2 δ⁻¹ k⌉` with `δ = ε/8`, i.e. `⌈16 k / ε⌉`. -/ noncomputable def ellOf (ε : ℝ) : ℕ := ⌈16 * kOf ε / ε⌉₊ /-- `ASM(P, ε, n)`: `k = ⌈8/ε⌉`, `δ = ε/8`, `ℓ = ⌈2k/δ⌉`, outer iterations `i = 0, …, ⌊log₂ n⌋`. -/ noncomputable def ASM (mm : MMOracle M W) (ε : ℝ) (n : ℕ) : St M W := P.asm (kOf ε) mm (ellOf ε) (Nat.log 2 n) end Prefs end
