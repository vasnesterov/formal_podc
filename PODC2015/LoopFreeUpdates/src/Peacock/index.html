import Papers.PODC2015.LoopFreeUpdates.PeacockCount /-! # Peacock (Algorithm 1) and Theorem 3 of Ludwig–Marcinkowski–Schmid (PODC 2015) `peacock I` is Algorithm 1, written recursively over the merged line: while the current line has at least two nodes, Peacock * (odd round, *shortcut*) updates the greedy selection `C = I.greedy` of far-reaching forward dashed edges (Algorithm 1, lines 6–9); * (even round, *prune*) updates all nodes not on the `s`–`d` path, `prune C` (line 11); * merges every updated node into the target of its new rule; the result is again a line (`reduce`, Lemma 3), on which Peacock continues. Theorem 3: the resulting schedule is relaxed loop-free (`peacock_isRLF`) and has at most `2 log_{3/2} n` rounds, `n` being the number of nodes (`peacock_length_le`). -/ set_option linter.unusedSectionVars false namespace Podc.LoopFreeUpdates open Relation variable {V : Type*} [DecidableEq V] namespace Instance /-- **Algorithm 1 (Peacock)**: the list of rounds `U₁, U₂, …`. -/ def peacock (I : Instance V) : List (Finset V) := if _h : I.old.length ≤ 1 then [] else I.greedy :: I.prune I.greedy :: peacock (I.reduce I.greedy_good) termination_by I.old.length decreasing_by exact I.greedy_good.reduce_length_lt (by omega) variable (I : Instance V) theorem peacock_of_le_one (h : I.old.length ≤ 1) : I.peacock = [] := by rw [peacock]; simp [h] theorem peacock_of_two_le (h : 2 ≤ I.old.length) : I.peacock = I.greedy :: I.prune I.greedy :: (I.reduce I.greedy_good).peacock := by rw [peacock]; simp [show ¬ I.old.length ≤ 1 by omega] theorem eq_d_of_length_le_one (h : I.old.length ≤ 1) {v : V} (hv : v ∈ I.old) : v = I.d := by have h1 := I.pos₁_lt hv have h2 := I.pos₁_d exact I.pos₁_inj hv I.d_mem (by omega) /-- The invariants of Peacock: rounds are disjoint sets of nodes, every node except the destination is updated, and every round is relaxed loop-free. -/ theorem peacock_spec : ∀ (n : ℕ) (I : Instance V), I.old.length = n → (∀ r ∈ I.peacock, r ⊆ I.nodes) ∧ I.peacock.Pairwise Disjoint ∧ (∀ v ∈ I.nodes, v ≠ I.d → ∃ r ∈ I.peacock, v ∈ r) ∧ (∀ (t : ℕ) (ht : t < I.peacock.length), ∀ X ⊆ I.peacock[t], I.RelaxedLoopFree (done I.peacock t ∪ X)) := by intro n induction n using Nat.strong_induction_on with | _ n ih => intro I hn by_cases h1 : I.old.length ≤ 1 · rw [I.peacock_of_le_one h1] refine ⟨by simp, by simp, fun v hv hvd => ?_, by simp⟩ exact absurd (I.eq_d_of_length_le_one h1 ((I.mem_nodes).1 hv)) hvd have h2 : 2 ≤ I.old.length := by omega set C := I.greedy have hC : I.GoodSelection C := I.greedy_good obtain ⟨ih1, ih2, ih3, ih4⟩ := ih _ (hn ▸ hC.reduce_length_lt h2) (I.reduce hC) rfl have hsub' : ∀ r ∈ (I.reduce hC).peacock, ∀ v ∈ r, v ∈ I.old ∧ I.Keep C v := by intro r hr v hv exact (hC.mem_reduce_old).1 (((I.reduce hC).mem_nodes).1 (ih1 r hr hv)) rw [I.peacock_of_two_le h2] refine ⟨?_, ?_, ?_, ?_⟩ · intro r hr v hv rw [I.mem_nodes] simp only [List.mem_cons] at hr rcases hr with rfl | rfl | hr · exact I.mem_old_of_fwd (hC.fwd v hv) · exact ((mem_prune).1 hv).1 · exact (hsub' r hr v hv).1 · refine List.Pairwise.cons ?_ (List.Pairwise.cons ?_ ih2) · intro r hr simp only [List.mem_cons] at hr rw [Finset.disjoint_left] rcases hr with rfl | hr · intro v hv hv'; linarith have h2 : 2 ≤ I.old.length := by omega have hC := I.greedy_good have hred := hC.reduce_length h2 have hih := ih _ (hn ▸ hC.reduce_length_lt h2) (I.reduce hC) rfl rw [I.peacock_of_two_le h2] simp only [List.length_cons, Nat.cast_add, Nat.cast_one] have hN' : (1 : ℝ) ≤ (I.reduce hC).old.length := by exact_mod_cast (I.reduce hC).length_pos have hle : ((I.reduce hC).old.length : ℝ) ≤ I.old.length * (3 / 2)⁻¹ := by have : (3 * (I.reduce hC).old.length + 1 : ℝ) ≤ 2 * I.old.length := by exact_mod_cast hred norm_num; linarith have hlog := Real.logb_le_logb_of_le (b := 3 / 2) (by norm_num) (by linarith) hle have hNpos : (0 : ℝ) < I.old.length := by exact_mod_cast hpos rw [Real.logb_mul hNpos.ne' (by norm_num), Real.logb_inv, Real.logb_self_eq_one (by norm_num)] at hlog linarith /-- **Theorem 3** (Ludwig–Marcinkowski–Schmid, PODC 2015): Peacock solves every instance with a relaxed loop-free schedule of `O(log n)` rounds;
