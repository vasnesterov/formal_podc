[4, 5] ++ -- W3 = W_z (refresh, the Byzantine server acks ⊥) [.invW (k+3)] ++ dWs allS ++ [.endW, .ack {0,1,2,3,4,5,6,7} (ba none)] ++ dRs [6] ++ dWs allS ++ [.endH] ++ -- R1 first iteration fails
