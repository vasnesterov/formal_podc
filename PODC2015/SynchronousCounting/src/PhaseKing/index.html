import Mathlib import Cslib.Computability.Distributed.Byzantine.KingPhase import Cslib.Foundations.Data.Finset.Quorum /-! # The phase-king layer (§3.4, Table 2, Lemmas 4 and 5) Every node `v` has an output register `a[v] ∈ [C] ∪ {∞}` (`Option (Fin C)`, `none = ∞`) and an auxiliary bit `d[v]`. In each round it executes one of three instruction sets (`I₃ℓ`, `I₃ℓ₊₁`, `I₃ℓ₊₂`, selected by `r % 3`) for a king `ℓ`, on the vector `ma` of received `a`-values. * `I₃ℓ`: if fewer than `N - F` nodes sent `a[v]`, `a[v] ← ∞`; increment `a[v]`. * `I₃ℓ₊₁`: `z_j` = number of received `j`; `d[v] ← [z_{a[v]} ≥ N - F]`; `a[v] ← min {j ∈ [C] : z_j > F}` (`∞` if there is none); increment `a[v]`. * `I₃ℓ₊₂`: if `a[v] = ∞` or `d[v] = 0`, `a[v] ← min {C, a[ℓ]}`; `d[v] ← 1`; increment `a[v]`. "Increment" is `+1 mod C` on `[C]` and does nothing on `∞`. The paper does not say what `min ∅` is in `I₃ℓ₊₁`; we use `∞`. In `I₃ℓ₊₂`, `min {C, ∞} = C` and `C + 1 mod C` is used literally. -/ namespace SynchronousCounting open Finset variable {V : Type*} [Fintype V] [DecidableEq V] {C : ℕ} /-- `x + 1 mod C` on `Fin C`. -/ def incr (x : Fin C) : Fin C := ⟨(x.val + 1) % C, Nat.mod_lt _ (Nat.zero_lt_of_lt x.isLt)⟩ /-- "increment a[v]". -/ def inc : Option (Fin C) → Option (Fin C) := Option.map incr /-- Number of received copies of the value `a`. -/ abbrev cntA (ma : V → Option (Fin C)) (a : Option (Fin C)) : ℕ := univ.countEq ma a /-- `min {j ∈ [C] : z_j > F}`, `∞` (`none`) if there is no such `j`. -/ def firstHeavy (F : ℕ) (ma : V → Option (Fin C)) : Option (Fin C) := (List.finRange C).find? fun j => decide (F < cntA ma (some j)) /-- `min {C, a}` for `a ∈ [C] ∪ {∞}`. -/ def kingVal : Option (Fin C) → ℕ | some x => x.val | none => C /-! ### Counting lemmas -/ variable (Fa : Finset V) /-- Values held by all correct nodes are received at least `N - |Fa|` times. -/ lemma cntA_of_all (ma : V → Option (Fin C)) (a : Option (Fin C)) (h : ∀ u ∉ Fa, ma u = a) : Fintype.card V - Fa.card ≤ cntA ma a := by simpa using card_sub_card_le_countEq (s := univ) (f := ma) fun u _ hu => h u hu omit [DecidableEq V] in /-- Values held by no correct node are received at most `|Fa|` times. -/ lemma cntA_of_none (ma : V → Option (Fin C)) (a : Option (Fin C)) (h : ∀ u ∉ Fa, ma u ≠ a) : cntA ma a ≤ Fa.card := by by_contra hlt obtain ⟨u, -, hu, hua⟩ := exists_mem_notMem_of_card_lt_countEq (not_le.1 hlt) exact h u hu hua omit [DecidableEq V] in lemma firstHeavy_eq_none (F : ℕ) (ma : V → Option (Fin C)) (h : ∀ j, cntA ma (some j) ≤ F) : firstHeavy F ma = none := by unfold firstHeavy rw [List.find?_eq_none] intro j _; simp only [decide_eq_true_eq, not_lt]; exact h j omit [DecidableEq V] in lemma firstHeavy_eq_some (F : ℕ) (ma : V → Option (Fin C)) (x : Fin C) (hx : F < cntA ma (some x)) (h : ∀ j, j ≠ x → cntA ma (some j) ≤ F) : firstHeavy F ma = some x := by unfold firstHeavy cases hfind : (List.finRange C).find?
