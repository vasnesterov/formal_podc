import Papers.PODC2015.HelpFreeWaitFree.Theorem48 /-! # Theorem 4.7: snapshots The snapshot object with components indexed by `K` (`update k v` writes `v` into component `k`, `scan` returns all components) is a strong global view type (`snapshot_strongGlobalView`), with the witnesses of the paper's proof of Theorem 4.7: `op = update k₁ 0`, the modifiers alternate `update k₂ 0`, `update k₂ 1`, … and the views are scans. Hence (`theorem_4_7`) there is no wait-free help-free linearizable snapshot implementation from read/write/cas registers for three processes. The paper states Theorem 4.7 for the *single-scanner* single-writer snapshot; our implementations must be correct for all executions (any process may invoke any operation), so this is the theorem for (multi-writer, multi-scanner) snapshot implementations. -/ set_option linter.unusedSectionVars false set_option linter.style.longLine false namespace Podc.HelpFreeWaitFree open Cslib.SharedMemory /-- Operations of a snapshot object with components `K` and values `V`. -/ inductive SnapOp (K V : Type) where | update (k : K) (v : V) | scan /-- The snapshot object: `update k v` writes `v` into component `k` (response `none`), `scan` returns the vector of all components (`none` = never written). -/ abbrev snapshotSpec (K V : Type) [DecidableEq K] : ObjSpec where State := K → Option V Op := SnapOp K V Resp := Option (K → Option V) init := fun _ => none δ q o := match o with | .update k v => (Function.update q k (some v), none) | .scan => (q, some q) section Snapshot variable {K : Type} [DecidableEq K] (k₁ k₂ : K) /-- The witnesses: `op = update k₁ true`, modifiers alternate `update k₂ true/false`, views are scans. -/ def snapOp : SnapOp K Bool := .update k₁ true /-- The modifiers. -/ def snapMod (n : ℕ) : SnapOp K Bool := .update k₂ (decide (n % 2 = 0)) /-- The snapshot state after a tagged history. -/ def snapState (y : List GVTag) : K → Option Bool := fun k => if k = k₁ then (if y.filter GVTag.isOp = [] then none else some true) else if k = k₂ then (match (y.filter GVTag.isMod).getLast? with | some (GVTag.mod n) => some (decide (n % 2 = 0)) | _ => none) else none lemma snap_runOps (h12 : k₁ ≠ k₂) (y : List GVTag) : (snapshotSpec K Bool).runOps (snapshotSpec K Bool).init (y.map (GVTag.toOp (snapOp k₁) (snapMod k₂) (fun _ => .scan))) = snapState k₁ k₂ y := by have key : ∀ q o, (snapshotSpec K Bool).runOps q [o] = ((snapshotSpec K Bool).δ q o).1 := fun _ _ => rfl induction y using List.reverseRecOn with | nil => funext k;
