given time. -/ namespace AsyncCycle variable {n : ℕ} {S O : Type*} (A : Alg S O) (init : ZMod n → S) (σ : ℕ → Finset (ZMod n)) /-- `p` fails (does not return) at time `t + 1`. -/ def FailsAt (t : ℕ) t p = Sum.inr o lemma failsAt_or_retAt {t : ℕ} {p : ZMod n} (h : ActsAt A init σ t p) : FailsAt A init σ t p ∨ RetAt A init σ t p := by rcases hr : resAt A init σ t p with s | o · exact Or.inl ⟨h, s, hr⟩ · exact Or.inr ⟨h, o, hr⟩ /-- After returning, a process never acts again. -/ lemma not_actsAt_of_retAt {t t' : ℕ} {p : ZMod n} (h : RetAt A init σ t p) (htt' : t < t') : ¬ ActsAt A init σ t' p := by obtain ⟨ha, o, ho⟩ := h have h1 := out_succ_of_inr A init σ ha ho have h2 := out_mono A init σ (Nat.succ_le_of_lt htt') h1 exact not_acts_of_out A init σ h2 /-- An activation followed by a later activation is a failure. -/ lemma failsAt_of_actsAt_later {t t' : ℕ} {p : ZMod n} (h : ActsAt A init σ t p) (htt' : t < t') (h' : ActsAt A init σ t' p) : FailsAt A init σ t p := by rcases
