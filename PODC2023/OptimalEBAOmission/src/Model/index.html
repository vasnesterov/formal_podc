import Mathlib import Cslib.Computability.Distributed.Knowledge.Basic /-! # The model of Alpturer–Halpern–van der Meyden (PODC 2023), §§2–6 *Optimal Eventual Byzantine Agreement Protocols with Omission Failures.* * **Information-exchange protocols** (§3). The local state of agent `i` is a pair `(core, x)`: `core = ⟨time, init, decided, jd⟩` is the part every EBA context must have (§5), `x : γ.X` is the rest. Messages are `Option γ.M` (`none` = `⊥`). `γ.tag : γ.M → Option Val` splits messages into `M⁰` (`tag = some 0`), `M¹` (`tag = some 1`) and `M²` (`tag = none`); deciding `v` sends an `Mᵛ`-message to everybody, `noop` sends `⊥` or `M²`-messages. The core part of the state is updated exactly as required by the definition of EBA contexts; the extra part by an arbitrary `γ.δX`. * **Failure patterns** `(N, F)`, `F m i j` = "the message from `i` to `j` in round `m + 1` is delivered"; a failure model is a set of patterns; `SO n t` is sending omissions with `≤ t` faulty agents. * **Runs.** A run is determined by its initial global state (`Config`: failure pattern, initial preferences, initial extra state) and the action protocol (`state P c m i` = `r_i(m)`). * **Knowledge** (`Knows P i s φ`): `φ` holds at every point of the system `I_{γ,P}` where `i` has local state `s` (the paper's `I, s ⊨ K_i φ`). * The knowledge-based program **`P0`**, **implementation**, the **EBA specification**, **domination** and **optimality** (§5), **0-chains** and **safety** (§6). Values `0, 1` are `Fin 2`; `1 - v` is the other value. -/ namespace Podc.OptimalEBAOmission open Classical /-- Binary values (initial preferences and decisions). -/ abbrev Val := Fin 2 /-- Actions: `decide v` or `noop`. -/ inductive Act where | decide (v : Val) | noop deriving DecidableEq /-- The components every local state of an EBA context has: `⟨time, init, decided, jd⟩` (`decided = none` is `⊥`; `jd = some v`: the agent learned that some agent just decided `v`). -/ @[ext] structure Core where /-- `time_i`. -/ time : ℕ /-- `init_i`. -/ init : Val /-- `decided_i ∈ {0, 1, ⊥}`. -/ decided : Option Val /-- `jd_i ∈ {0, 1, ⊥}`. -/ jd : Option Val deriving DecidableEq /-- A failure pattern `(N, F)`: `N` is the set of nonfaulty agents and `F m i j` says whether the message from `i` to `j` in round `m + 1` is delivered. -/ structure Pattern (n : ℕ) where /-- The nonfaulty agents. -/ N : Finset (Fin n) /-- Delivery: `F m i j` for the message from `i` to `j` in round `m + 1`. -/ F : ℕ → Fin n → Fin n → Bool /-- The sending-omissions failure model `SO(t)`: at most `t` faulty agents, and only faulty agents omit messages. -/ def SO (n t : ℕ) : Set (Pattern n) := {α | (Finset.univ \ α.N).card ≤ t ∧ ∀ m i j, α.F m i j = false → i ∉ α.N} /-- An **EBA context** for `n` agents (§5): an information-exchange protocol satisfying the EBA constraints, together with a failure model. (The interpretation `π` of the paper is not needed: formulas are semantic predicates.) -/ structure Context (n : ℕ) where /-- The non-EBA part of the local states. -/ X : Type /-- Messages (`none : Option M` is `⊥`). -/ M : Type /-- `tag m = some v` iff `m ∈ Mᵛ` (`v ∈ {0,1}`); `tag m = none` iff `m ∈ M²`. -/ tag : M → Option Val /-- Initial values of the non-EBA part, for agent `i` with initial preference `v`. -/ initX : Fin n → Val → Set X /-- Every agent can start with either preference ("independently selected value init_i"). -/ initX_nonempty : ∀ i v, (initX i v).Nonempty /-- Message selection `μ_i(s, a)(j)`. -/ μ : Fin n → Core × X → Act → Fin n → Option M /-- Deciding `v` sends an `Mᵛ`-message to every agent. -/ μ_decide : ∀ i s v j, ∃ m, μ i s (.decide v) j = some m ∧ tag m = some v /-- Not deciding sends `⊥` or an `M²`-message. -/ μ_noop : ∀ i s j m, μ i s .noop j = some m → tag m = none /-- Update of the non-EBA part of the state. -/ δX : Fin n → Core × X → Act → (Fin n → Option M) → X /-- The failure model. -/ FM : Set (Pattern n) namespace Context variable {n : ℕ} (γ : Context n) /-- Local states. -/ abbrev L := Core × γ.X /-- Action protocols `P_i : L_i → A_i`. -/ abbrev Protocol := Fin n → γ.L → Act /-- The new value of `decided` after action `a`. -/ def newDec (d : Option Val) : Act → Option Val | .decide v => some v | .noop => d /-- The new value of `jd`: `0` if a message from an agent deciding `0` was received, else `1` if a message from an agent deciding `1` was received, else `⊥`. (The paper does not say what happens if both are received; this never happens with implementations of `P0`.) -/ noncomputable def jdOf (rcv : Fin n → Option γ.M) : Option Val := if ∃ j m, rcv j = some m ∧ γ.tag m = some 0 then some 0 else if ∃ j m, rcv j = some m ∧ γ.tag m = some 1 then some 1 else none /-- The state update `δ_i`: time increases by one, `init` is kept, `decided` records a decision, `jd` records what was just decided; the rest is updated by `δX`. -/ noncomputable def δ (i : Fin n) (s : γ.L) (a : Act) (rcv : Fin n → Option γ.M) : γ.L := (⟨s.1.time + 1, s.1.init, newDec s.1.decided a, γ.jdOf rcv⟩, γ.δX i s a rcv) /-- The initial global state of a run: a failure pattern, initial preferences and the initial values of the non-EBA parts of the local states. -/ structure Config where /-- The failure pattern (state of the environment). -/ pat : Pattern n /-- Initial preferences. -/ init : Fin n → Val /-- Initial extra state. -/ x0 : Fin n → γ.X variable {γ} /-- A configuration is admissible if its pattern is in the failure model and the initial local states are initial states of the information-exchange protocol. -/ def Config.Valid (c : γ.Config) : Prop := c.pat ∈ γ.FM ∧ ∀ i, c.x0 i ∈ γ.initX i (c.init i) variable (γ) /-- The run of protocol `P` from configuration `c`: `state P c m i = r_i(m)`. -/ noncomputable def state (P : γ.Protocol) (c : γ.Config) : ℕ → Fin n → γ.L | /-- `deciding_j = v` at `(c, m)`: `decided_j = ⊥ ∧ ◯ decided_j = v` ("`j` decides `v` in round `m + 1`"). -/ def Deciding (P : γ.Protocol) (c : γ.Config) (m : ℕ) (j : Fin n) (v : Val) : Prop := dec P c m j = none ∧ dec P c (m + 1) j = some v /-- `jdecided_j = v` at `(c, m)`: `decided_j = v ∧ ⊖ decided_j = ⊥` (`j` just decided `v`). -/ def JDecided (P : γ.Protocol) (c : γ.Config) (m : ℕ) (j : Fin n) (v : Val) : Prop := 0 < m ∧ dec P c m j = some v ∧ dec P c (m - 1) j = none /-- `I_{γ,P}, s ⊨ K_i φ`: `φ` holds at every point of the system of `P` where `i` has local state `s`. -/ def Knows (P : γ.Protocol) (i : Fin n) (s : γ.L) (φ : γ.Config → ℕ → Prop) : Prop := ∀ c : γ.Config, c.Valid → ∀ m, γ.state P c m i = s → φ c m /-- The knowledge-based program `P0` (§6), with its tests interpreted in the system of `P`: ``` if decided_i ≠ ⊥ then noop else if init_i = 0 ∨ K_i (⋁_j jdecided_j = 0) then decide_i(0) else if K_i (⋀_j ¬(deciding_j = 0)) then decide_i(1) else noop ``` -/ noncomputable def P0 (P : γ.Protocol) : γ.Protocol := fun i s => if s.1.decided ≠ none then .noop else if s.1.init = 0 ∨ Knows P i s (fun c m => ∃ j, JDecided P c m j 0) then .decide 0 else if Knows P i s (fun c m => ∀ j, ¬ Deciding P c m j 0) then .decide 1 else .noop /-- `P` implements the knowledge-based program `P0` in `γ`: `P_i(s) = P0_i^{I_{γ,P}}(s)` for all local states `s` that arise in the system of `P`. -/ def Implements (P : γ.Protocol) : Prop := ∀ c : γ.Config, c.Valid → ∀ m i, P i (γ.state P c m i) = P0 P i (γ.state P c m i) /-- `P` is an **EBA decision protocol** for `γ` (§5): Unique Decision, Agreement, Validity and Termination are valid in `I_{γ,P}`. -/ structure IsEBA (P : γ.Protocol) : Prop where /-- `decided_i = v ⇒ □ ¬(decided_i = 1 - v)`. -/ unique : ∀ c : γ.Config, c.Valid → ∀ i m v, dec P c m i = some v → ∀ m' ≥ m, dec P c m' i ≠ some (1 - v) /-- `¬(i ∈ N ∧ j ∈ N ∧ decided_i = v ∧ decided_j = v')` for `v ≠ v'`. -/ agreement : ∀ c : γ.Config, c.Valid → ∀ i ∈ c.pat.N, ∀ j ∈ c.pat.N, ∀ m v v', dec P c m i = some v → dec P c m j = some v' → v = v' /-- `(decided_i = v ∧ i ∈ N) ⇒ ∃v`. -/ validity : ∀ c : γ.Config, c.Valid → ∀ i ∈ c.pat.N, ∀ m v, dec P c m i = some v → ∃ j, c.init j = v /-- `i ∈ N ⇒ ◇ (decided_i ≠ ⊥)`. -/ termination : ∀ c : γ.Config, c.Valid → ∀ i ∈ c.pat.N, ∀ m, ∃ m' ≥ m, dec P c m' i ≠ none /-- `Dominates P P'` is the paper's `P' ≤_γ P` ("`P` dominates `P'`"): in corresponding runs, if a nonfaulty `i` performs `decide_i(v)` at time `m` with `P`, then `P'` does not perform any `decide_i(w)` at a time `m' < m`. -/ def Dominates (P P' : γ.Protocol) : Prop := ∀ c : γ.Config, c.Valid → ∀ i ∈ c.pat.N, ∀ m v, P i (γ.state P c m i) = .decide v → ∀ m' < m, ∀ w, P' i (γ.state P' c m' i) ≠ .decide w /-- `P` strictly dominates `P'`: `P' ≤_γ P` and not `P ≤_γ P'`. -/ def StrictlyDominates (P P' : γ.Protocol) : Prop := Dominates P P' ∧ ¬ Dominates P' P /-- `P` is **optimal** for `γ`: an EBA decision protocol not strictly dominated by any EBA decision protocol. -/ def Optimal (P : γ.Protocol) : Prop := IsEBA P ∧ ∀ P', IsEBA P' → ¬ StrictlyDominates P' P /-- A protocol that performs no action once it has decided (the decision is a one-shot action).
