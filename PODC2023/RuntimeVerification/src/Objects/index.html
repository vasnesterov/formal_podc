Operations carry a fictitious tag so that a process never invokes the same operation twice (Section 2). ## Queue -/ inductive QOp | enq (v tag : ℕ) | deq (tag : ℕ) deriving DecidableEq inductive QRes | ok | val (v : ℕ) | empty deriving DecidableEq /-- FIFO queue. -/ def queueδ : List ℕ → QOp → List ℕ × QRes | q, .enq v _ => (q ++ [v], .ok) | [], .deq _ => ([], .empty) | v :: q, .deq _ => (q, .val v) def queueData : ImpData QOp QRes := ⟨.enq 1 0, .deq 0, fun k => .deq (k + 1), .ok, .val 1, .empty⟩ theorem queue_good (hn : 2 ≤ n) : queueData.Good hn (fun _ => detSpec [] queueδ) := good_of_det hn [] queueδ queueData [1] [] rfl rfl (fun _ => rfl) (by decide) (fun _ _ h => by simpa [queueData] using h) (fun _ => by simp [queueData]) (fun _ => by simp [queueData]) /-- **Theorem 5.1 for queues.** -/ theorem thm_5_1_queue (hn : 2 ≤ n) (V : Verifier n QOp QRes (Hist (Fin n) QOp QRes)) : ¬ RuntimeVerifies V {H | Linearizable (fun _ => detSpec [] queueδ) H} := thm_5_1 (queue_good hn) V /-- **Theorem A.1 for queues.** -/ theorem thm_A_1_queue (hn : 2 ≤ n) (V : Verifier n QOp QRes (Hist (Fin n) QOp QRes)) : ¬ PredictivelyVerifies V {H | Linearizable (fun _ => detSpec [] queueδ) H} := thm_A_1 (queue_good hn) V /-!
