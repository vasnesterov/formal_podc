import Papers.PODC2023.DuraCAS.Spec /-! # DuraCAS is not durably linearizable under the brief announcement's recovery discipline The brief announcement states the obligation of a crashed process *per object*: "when `π` subsequently restarts, it must execute `O.Recover()` before invoking any other operation on `O`". So after crashing inside an operation on `O₀`, a process may invoke an operation on another DuraCAS object `O₁` first. But the DurEC handle `h.Critical` (with its `Val` field) is shared by all objects: if the crash happened right after the install at line 10 of DurEC's `ECSC` on `O₀`'s `Z` (the install is not forwarded yet), the `ECSC` of the new operation on `O₁` overwrites `h.Critical.Val` at line 7, and any helper then forwards *that* value into `O₀`'s `Z`. `StepPO` adds the transition allowed by the per-object discipline ("postpone the recovery of `O` and invoke an operation on `O' ≠ O`"). `perObject_counterexample`: two processes, two objects, initial values `0`; a `READ` of `O₀` returns `2`, a value only ever proposed to `O₁`. -/ namespace Podc.DuraCAS variable {P K V : Type} [DecidableEq P] [DecidableEq K] [DecidableEq V] /-- Labels of the per-object discipline: additionally, a process that has just crashed (it is at line 22 of `RECOVER` of its pending operation on `k`) may invoke an operation on `k' ≠ k` instead. -/ inductive LblPO (P K V : Type) where | base (ℓ : Lbl P K V) | postpone (p : P) (k : K) (op : Op V) deriving DecidableEq /-- Enabledness under the per-object discipline. -/ def LblPO.En (s : St P K V) : LblPO P K V → Prop | .base ℓ => ℓ.En s | .postpone p k _ => (s.ps p).pc = .recv 0 ∧ (s.ps p).fr = none ∧ ((s.ps p).op.map OpRec.k).isSome ∧ (s.ps p).op.map OpRec.k ≠ some k instance (s : St P K V) : (ℓ : LblPO P K V) → Decidable (ℓ.En s) | .base ℓ => inferInstanceAs (Decidable (ℓ.En s)) | .postpone p k _ => inferInstanceAs (Decidable ((s.ps p).pc = .recv 0 ∧ (s.ps p).fr = none ∧ ((s.ps p).op.map OpRec.k).isSome ∧ (s.ps p).op.map OpRec.k ≠ some k)) /-- Effect of a per-object-discipline transition. -/ def LblPO.apply (s : St P K V) : LblPO P K V → St P K V | .base ℓ => ℓ.apply s | .postpone p k op => doInvoke s p k op /-- One step under the per-object recovery discipline. -/ def StepPO (s s' : St P K V) : Prop := ∃ ℓ : LblPO P K V, ℓ.En s ∧ s' = ℓ.apply s /-- Reachable states under the per-object discipline. -/ def ReachPO (init : K → V) (hv0 : Hd P → V × Bool) : St P K V → Prop := Relation.ReflTransGen StepPO (initSt init hv0) /-- Run a list of labels (none if some label is not enabled). -/ def runPO (s : St P K V) : List (LblPO P K V) → Option (St P K V) | [] => some s | ℓ :: ls => if ℓ.En s then runPO (ℓ.apply s) ls else none theorem reachPO_of_runPO {init : K → V} {hv0} {s s' : St P K V} (hs : ReachPO init hv0 s) : ∀ ls, runPO s ls = some s' → ReachPO init hv0 s' := by intro ls induction ls generalizing s with | nil => intro h;
