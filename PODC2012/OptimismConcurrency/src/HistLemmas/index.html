import Papers.PODC2012.OptimismConcurrency.ImplModel /-! # Lemmas on histories (extension by one event) How `clientSt`, the operations of a process (`splitOps`, `restrictPX`), the high-level history `hl`, and the runs `seqRun`, `honRun` behave when one event is appended to a history. -/ namespace Podc.OptimismConcurrency open Cslib.SharedMemory section General variable {P X Op Res Obj Val : Type} /-- The step function of `splitOps`. -/ def splitF (e : Ev P X Op Res Obj Val) (acc : Hist P X Op Res Obj Val × List (Hist P X Op Res Obj Val)) : Hist P X Op Res Obj Val × List (Hist P X Op Res Obj Val) := if e.isInv then ([], (e :: acc.1) :: acc.2) else (e :: acc.1, acc.2) lemma splitOps_eq (l : Hist P X Op Res Obj Val) : splitOps l = (l.foldr splitF ([], [])).2 := rfl lemma modifyLast_id' {α : Type*} (S : List α) : S.modifyLast (fun x => x) = S := by rcases List.eq_nil_or_concat S with rfl | ⟨L, b, rfl⟩ · rfl · rw [List.concat_eq_append, List.modifyLast_concat] /-- Attach the partial segment `a` to the last segment, and the segments `b` after. -/ def glue (a : Hist P X Op Res Obj Val) (b : List (Hist P X Op Res Obj Val)) : Hist P X Op Res Obj Val × List (Hist P X Op Res Obj Val) → Hist P X Op Res Obj Val × List (Hist P X Op Res Obj Val) | (f, []) => (f ++ a, b) | (f, S) => (f, S.modifyLast (· ++ a) ++ b) lemma splitF_glue (x : Ev P X Op Res Obj Val) (a : Hist P X Op Res Obj Val) (b : List (Hist P X Op Res Obj Val)) (fs : Hist P X Op Res Obj Val × List (Hist P X Op Res Obj Val)) : splitF x (glue a b fs) = glue a b (splitF x fs) := by rcases fs with ⟨f, S⟩ cases S with | nil => by_cases hx : x.isInv · simp only [splitF, glue, hx, ↓reduceIte] rw [show [x :: f] = [] ++ [x :: f] from rfl, List.modifyLast_concat];
