simpa using hwf | cons ℓ l ih => intro α s st L c' h hwf hrun rw [lpRun_cons] at hrun cases hstep : lpStep (s, st) ℓ with | none => rw [hstep] at hrun; simp at hrun | some c₁ => rw [hstep, Option.bind_some] at hrun obtain ⟨L₁, h₁⟩ := h.step hstep have hwf₁ : WellFormed (α ++ (ℓ.erase).toList) := by cases ℓ with | inv p o => simp only [lpStep] at hstep split at hstep <;> try simp at hstep rename_i hp have hsp := h.status p; rw [hp] at hsp exact hwf.append_inv hsp | ret p r => simp only [lpStep] at hstep split at hstep <;> try simp at hstep rename_i r' hp have hsp := h.status p; rw [hp] at hsp exact hwf.append_ret hsp.1 | lin p => simpa [LPLabel.erase] using hwf have := ih h₁ hwf₁ hrun cases he : ℓ.erase <;> simp_all [List.filterMap_cons] /-- The checker only accepts well-formed histories. -/ theorem wellFormed_of_lpRun {l : List (LPLabel P A)} {c' : A.State × (P → LPStatus A)} (h : lpRun (A.init, fun _ => .idle) l = some c') : WellFormed (l.filterMap LPLabel.erase) := by simpa using LPInv.wellFormed_of_lpRun lpInv_nil (fun β ℓ γ he => by simp at he) h omit [DecidableEq A.Resp] in /-- Erasures of annotated histories contain no internal steps, so `ofHist ∘ toHist` is the identity on them. -/ lemma ofHist_toHist_erase (l : List (LPLabel P A)) : HerlihyWing.ofHist (HerlihyWing.toHist (l.filterMap LPLabel.erase)) = l.filterMap LPLabel.erase := by induction l with | nil => rfl | cons ℓ l ih => cases ℓ with | inv p o => simpa [LPLabel.erase, HerlihyWing.toHist, HerlihyWing.ofHist, HerlihyWing.labelEvents, HerlihyWing.eventLabel] using ih | ret p r => simpa [LPLabel.erase, HerlihyWing.toHist, HerlihyWing.ofHist, HerlihyWing.labelEvents, HerlihyWing.eventLabel] using ih | lin p => simp only [List.filterMap_cons, LPLabel.erase];
