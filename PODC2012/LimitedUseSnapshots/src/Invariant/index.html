import Papers.PODC2012.LimitedUseSnapshots.Semantics import Cslib.Computability.Distributed.SharedMemory.Linearization /-! # The invariant of the corrected Algorithm 2 Definitions of the invariant `Inv α C` (execution `α` of `alg P T true` with final configuration `C`) used to prove that the corrected algorithm satisfies the max-array linearizability criterion of `Criterion.lean`, and its basic stability lemmas. The invariant follows the paper's proof (Lemmas 3.1–3.3, Theorem 3.4) and the erratum, flattened over the whole tree: * `g1` — `second` of a child is at most `second` of its parent (Lemma 3.2); * per-process invariants `PInv`: a scan at node `a` has a lower bound `lb` on its future second component (Lemma 3.3), its path is consistent with the switches at its invocation (`LeftDom`) and with the current switches (`RightDom`), every right step `u` of its path is *guarded* (`Guard`: `u.left.second` and all pending left propagations at `u` are `≤ lb`; Lemma 3.1), and it carries a candidate *witness* `MaxUpdate0` for its first component; * `q` — every value of `second` at a node `c` was the value of the root's `second` at a time when the path to `c` was consistent with the switches; * `fs` — every switch that is on was turned on by a `MaxUpdate0` whose descent below the switch is complete (`Below`), and whose path above it was consistent with the switches at its invocation; * `comp`, `compScan`, `hist` — what is known about completed operations. -/ set_option linter.unusedSectionVars false namespace Podc.LimitedUseSnapshots open Cslib.SharedMemory variable {P : Type} [DecidableEq P] (T : Shape) /-- `ℓ` is the address of a leaf. -/ def IsLeaf (ℓ : List Bool) : Prop := T.IsAddr ℓ ∧ T.subD ℓ = .leaf /-- The descent of `MaxUpdate0(v)` below node `c` (on the path to `v`) is complete: the switches of all right steps down to some node `e` are on, and `e` is the leaf `v`, or a left step whose switch is on (where the descent stopped). -/ def Below (C : Cfg P) (c : List Bool) (v : ℕ) : Prop := ∃ e, c <+: e ∧ e <+: T.addr v ∧ (∀ u x, e = u ++ true :: x → c <+: u → sw C u = true) ∧ (T.subD e = .leaf ∨ (sw C e = true ∧ e ++ [false] <+: T.addr v)) /-- Some leaf reachable through switches that are on has `second ≥ g`. -/ def Good1 (C : Cfg P) (g : ℕ) : Prop := ∃ ℓ, IsLeaf T ℓ ∧ RightDom ℓ (sw C) ∧ g ≤ sc C ℓ /-- The left child of `u` and all pending left propagations at `u` are `≤ b`. -/ def Guard (C : Cfg P) (u : List Bool) (b : ℕ) : Prop := sc C (u ++ [false]) ≤ b ∧ ∀ q e y, C.loc q = .s3 e u y → y ≤ b /-- Every right step of the path `c` is guarded
