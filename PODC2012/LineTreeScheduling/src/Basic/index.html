import Mathlib /-! # Throughput maximization on line and tree networks: the abstract problem Chakaravarthy, Roy, Sabharwal, *Distributed algorithms for scheduling on line and tree networks*, PODC 2012 (arXiv:1205.1924v2). After the reformulation of Section 2 every problem considered in the paper (tree networks, line networks with windows, unit or arbitrary heights) is a packing problem over a finite set `D` of *demand instances*: each `d : D` belongs to a demand `dem d : A`, has a profit, and uses a finite set `path d : Finset E` of edges. Edges of different networks are different elements of `E`, so two instances *overlap* iff their edge sets intersect. * `Inst A D E` — the abstract instance. * `Overlap`, `Conflict`, `IsIndep` — Section 2 (a feasible solution of the unit-height problem is an independent set of demand instances). * `IsHFeasible` — feasible solutions with heights (Section 6). * `IsLPFeasible h x` — the LP relaxation of Section 3.1 (`h = 1`) and of Section 6.1 (heights `h`). * `Duals`, `lhs`, `val`, `weak_duality` — the dual LP and weak duality, in the scaled form used by Lemmas 3.1 and 6.1 (if all dual constraints are `λ`-satisfied then `λ · (LP value) ≤ val`). -/ set_option linter.unusedSectionVars false namespace Podc.LineTreeScheduling open Finset /-- An instance of the throughput maximization problem, reformulated in terms of demand instances (Section 2): `dem d` is the demand `a_d` owning the instance `d`, `path d` the set of edges used by `d`, `profit d` its profit. -/ structure Inst (A D E : Type*) where /-- The demand `a_d` to which the demand instance `d` belongs. -/ dem : D → A /-- The edges of `path(d)`. -/ path : D → Finset E /-- The profit `p(d)`. -/ profit : D → ℝ variable {A D E : Type*} [DecidableEq A] [DecidableEq D] [DecidableEq E] namespace Inst variable (I : Inst A D E) /-- `d₁` and `d₂` overlap: their paths share an edge. -/ def Overlap (d₁ d₂ : D) : Prop := (I.path d₁ ∩ I.path d₂).Nonempty /-- `d₁` and `d₂` conflict: they belong to the same demand or they overlap. -/ def Conflict (d₁ d₂ : D) : Prop := I.dem d₁ = I.dem d₂ ∨ I.Overlap d₁ d₂ /-- An independent set of demand instances (= a feasible solution in the unit-height case). -/ def IsIndep (S : Finset D) : Prop := ∀ d₁ ∈ S, ∀ d₂ ∈ S, d₁ ≠ d₂ → ¬ I.Conflict d₁ d₂ /-- The profit `p(S)` of a set of demand instances. -/ def profitOf (S : Finset D) : ℝ := ∑ d ∈ S, I.profit d lemma overlap_comm {d₁ d₂ : D} : I.Overlap d₁ d₂ ↔ I.Overlap d₂ d₁ := by unfold Overlap; (With `λ = 1` this is the usual weak duality theorem; the paper scales `(α, β)` by `1/λ` instead.) -/ theorem weak_duality {h : D → ℝ} {x : D → ℝ} (hx : I.IsLPFeasible h x) {y : Duals A E} (hα : ∀ a, 0 ≤ y.α a) (hβ : ∀ e, 0 ≤ y.β e) {lam : ℝ} (hsat : ∀ d, lam * I.profit d ≤ I.lhs h y d) : lam * I.lpValue x ≤ y.val := by classical obtain ⟨hx0, hxe, hxa⟩ := hx have h1 : lam * I.lpValue x ≤ ∑ d, x d * I.lhs h y d := by rw [lpValue, Finset.mul_sum] refine Finset.sum_le_sum fun d _ => ?_ have := mul_le_mul_of_nonneg_left (hsat d) (hx0 d) linarith have h2 : ∑ d, x d * I.lhs h y d = ∑ a, y.α a * ∑ d ∈ univ.filter (fun d => I.dem d = a), x d + ∑ e, y.β e * ∑ d ∈ univ.filter (fun d => e ∈ I.path d), x d * h d := by simp only [lhs, mul_add, Finset.sum_add_distrib] congr 1 · rw [← Finset.sum_fiberwise univ I.dem (fun d => x d * y.α (I.dem d))] refine Finset.sum_congr rfl fun a _ => ?_ rw [Finset.mul_sum] refine Finset.sum_congr rfl fun d hd => ?_ rw [(Finset.mem_filter.1 hd).2]; ring · simp_rw [Finset.mul_sum] rw [Finset.sum_comm' (t' := univ) (s' := fun e => univ.filter (fun d => e ∈ I.path d))] · refine Finset.sum_congr rfl fun e _ => Finset.sum_congr rfl fun d _ => by ring · intro d e; simp have h3 : ∑ a, y.α a * ∑ d ∈ univ.filter (fun d => I.dem d = a), x d ≤ ∑ a, y.α a := Finset.sum_le_sum fun a _ => mul_le_of_le_one_right (hα a) (hxa a) have h4 : ∑ e, y.β e * ∑ d ∈ univ.filter (fun d => e ∈ I.path d), x d * h d ≤ ∑ e, y.β e := Finset.sum_le_sum fun e _ => mul_le_of_le_one_right (hβ e) (hxe e) rw [Duals.val]; linarith end Inst end Podc.LineTreeScheduling
