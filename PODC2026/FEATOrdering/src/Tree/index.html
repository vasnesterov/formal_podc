import Papers.PODC2026.FEATOrdering.Fairness /-! # Comparator-guided BST insertion and two-pointer merge (Lemmas 6.1, 6.2) FEAT builds, in each round, a balanced binary search tree whose branching is guided by a (possibly non-transitive) comparator, and takes its in-order traversal; it then stitches the result with the tail list by a two-pointer merge. * `BTree.insert c x t` — Algorithm 2 (TryInsert) without the `⊥` case: walk down from the root, going left at `u` iff `c x u = True`, and attach `x` as a leaf. * `buildTree c bal xs` — insert the transactions `xs` in this order; after every insertion apply `bal` (AVL rebalancing; we only assume that it preserves the in-order traversal, which holds for rotations). * `lemma_6_1` — the in-order traversal satisfies Adjacent-Comparator Consistency for every tournament comparator, every insertion order and every in-order-preserving rebalancing. * `lemma_6_2` — the two-pointer merge (`List.merge`, which is exactly the paper's stitching loop once `cmp` is known) of two consistent lists is consistent. -/ namespace Podc.FEATOrdering variable {α : Type*} /-- Binary trees with keys at internal nodes. -/ inductive BTree (α : Type*) where | nil : BTree α | node (l : BTree α) (x : α) (r : BTree α) : BTree α namespace BTree /-- In-order traversal. -/ def inorder : BTree α → List α | nil => [] | node l x r => inorder l ++ x :: inorder r /-- Comparator-guided BST insertion (Algorithm 2 with a total comparator): go left at `y` iff `c x y = True`, attach `x` as a new leaf. -/ def insert (c : α → α → Bool) (x : α) : BTree α → BTree α | nil => node nil x nil | node l y r => if c x y then node (insert c x l) y r else node l y (insert c x r) /-- Inserting `x` puts it at exactly one position of the in-order traversal, between its in-order predecessor `p` (with `c x p = False`: the search went right at `p`) and its successor `s` (with `c x s = True`: the search went left at `s`). -/ theorem inorder_insert (c : α → α → Bool) (x : α) (t : BTree α) : ∃ L₁ L₂, t.inorder = L₁ ++ L₂ ∧ (t.insert c x).inorder = L₁ ++ x :: L₂ ∧ (∀ p ∈ L₁.getLast?, c x p = false) ∧ (∀ s ∈ L₂.head?, c x s = true) := by induction t with | nil => exact ⟨[], [], rfl, rfl, by simp, by simp⟩ | node l y r ihl ihr => by_cases hxy : c x y = true · obtain ⟨A₁, A₂, h1, h2, h3, h4⟩ := ihl refine ⟨A₁, A₂ ++ y :: r.inorder, by simp [inorder, h1], ?_, h3, ?_⟩ · simp [insert, hxy, inorder, h2] · intro s hs cases A₂ with | nil => simp at hs; exact hs ▸ hxy | cons a A₂ => exact h4 s (by simpa using hs) · obtain ⟨B₁, B₂, h1, h2, h3, h4⟩ := ihr refine ⟨l.inorder ++ y :: B₁, B₂, by simp [inorder, h1], ?_, ?_, h4⟩ · simp [insert, hxy, inorder, h2] · intro p hp rcases List.eq_nil_or_concat B₁ with h | ⟨B, b, h⟩ · subst h; simp at hp; exact hp ▸ (by simpa using hxy) · subst h rw [show l.inorder ++ y :: B.concat b = (l.inorder ++ y :: B) ++ [b] by simp, List.getLast?_concat] at hp have : p = b := by simpa using hp.symm exact h3 p (by simp [this, List.concat_eq_append]) theorem inorder_insert_perm (c : α → α → Bool) (x : α) (t : BTree α) : (t.insert c x).inorder.Perm (x :: t.inorder) := by obtain ⟨L₁, L₂, h1, h2, -, -⟩ := inorder_insert c x t rw [h1, h2] exact List.perm_middle end BTree /-- Build the round tree: insert the transactions of `xs` in order, rebalancing with `bal` after each insertion. -/ def buildTree (c : α → α → Bool) (bal : BTree α → BTree α) (xs : List α) : BTree α := xs.foldl (fun t x => bal (t.insert c x)) .nil /-- A comparator is *total* if for all `a, b` one of `c a b`, `c b a` holds (a tournament with `c a a = True`). -/ def TotalCmp (c : α → α → Bool) : Prop := ∀ a b, c a b = true ∨ c b a = true theorem AdjConsistent.insert {c : α → α → Bool} (htot : TotalCmp c) {t : BTree α} (x : α) (h : AdjConsistent c t.inorder) : AdjConsistent c (t.insert c x).inorder := by obtain ⟨L₁, L₂, h1, h2, h3, h4⟩ := BTree.inorder_insert c x t unfold AdjConsistent at h ⊢ rw [h1] at h rw [h2] obtain ⟨hL₁, hL₂, -⟩ := List.isChain_append.1 h refine List.isChain_append.2 ⟨hL₁, List.isChain_cons.2 ⟨h4, hL₂⟩, ?_⟩ intro p hp y hy simp at hy; exact ha ▸ hzx x (by simp), ?_⟩ exact isChain_cons_merge htot xs (y :: ys) x (List.isChain_cons.1 hx).2 hy (List.isChain_cons.1 hx).1 (by simpa using hxy) · rw [if_neg hxy] have hyx : c y x = true := (htot x y).resolve_left hxy refine List.isChain_cons.2 ⟨fun a ha => by simp at ha;
