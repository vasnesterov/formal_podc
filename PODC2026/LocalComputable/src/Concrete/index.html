import Papers.PODC2026.LocalComputable.Basic import Mathlib.Data.Finset.Card import Mathlib.Algebra.Order.Monoid.Unbundled.Pow /-! # The concrete LOCAL model of Section 3 Networks `(G, x, deg, id, {p_v})` of Section 3: finite simple graphs of maximum degree `Δ`, port numbering (the order of each node's neighbour list), unique identifiers in `[1, n^c]`, and inputs from a finite alphabet `{0, …, k-1}`. Nodes are `0, …, n-1`. Views: the radius-`t` *full-information view* of `v` — what `v` knows after `t` rounds of unbounded message passing: its identifier, input and degree and, for `t > 0`, the radius-`(t-1)` views of its neighbours in port order. With unique identifiers this carries the same information as the rooted labelled neighbourhood `N^t(v)` of Definition 3.1 (nodes at distance `≤ t` with their degrees, edges with an endpoint at distance `≤ t-1`, ports, identifiers, inputs). The abstract results only use the restriction property (`fview_restrict`), which holds for both. -/ namespace LocalComputable /-- A raw port-numbered network with identifiers and inputs (`nbr[v]` = neighbours of `v` in port order). -/ structure PNet where /-- number of nodes -/ n : ℕ /-- port-ordered neighbour lists -/ nbr : List (List ℕ) /-- identifiers -/ id : List ℕ /-- local inputs -/ inp : List ℕ deriving DecidableEq namespace PNet /-- Neighbours of `v` in port order. -/ def nbrOf (G : PNet) (v : ℕ) : List ℕ := G.nbr.getD v [] /-- Identifier of `v`. -/ def idOf (G : PNet) (v : ℕ) : ℕ := G.id.getD v 0 /-- Input of `v`. -/ def inpOf (G : PNet) (v : ℕ) : ℕ := G.inp.getD v 0 /-- Well-formed networks of Section 3: simple undirected graphs of maximum degree `Δ`, unique identifiers in `[1, n^c]`, inputs in `{0, …, k-1}`. -/ structure WF (Δ c k : ℕ) (G : PNet) : Prop where len_nbr : G.nbr.length = G.n len_id : G.id.length = G.n len_inp : G.inp.length = G.n nodup : ∀ v < G.n, (G.nbrOf v).Nodup deg_le : ∀ v < G.n, (G.nbrOf v).length ≤ Δ nbr_lt : ∀ v < G.n, ∀ u ∈ G.nbrOf v, u < G.n irrefl : ∀ v < G.n, v ∉ G.nbrOf v symm : ∀ v < G.n, ∀ u ∈ G.nbrOf v, v ∈ G.nbrOf u id_inj : ∀ u < G.n, ∀ v < G.n, G.idOf u = G.idOf v → u = v id_pos : ∀ v < G.n, 1 ≤ G.idOf v id_le : ∀ v < G.n, G.idOf v ≤ G.n ^ c inp_lt : ∀ v < G.n, G.inpOf v < k end PNet /-- Full-information views: identifier, input, VTree open PNet VTree /-- The radius-`t` full-information view of `v` in `G`. -/ def fview (G : PNet) : ℕ → ℕ → VTree | v, 0 => node (G.idOf v) (G.inpOf v) (G.nbrOf v).length [] | v, t + 1 => node (G.idOf v) (G.inpOf v) (G.nbrOf v).length ((G.nbrOf v).map fun u => fview G u t) theorem trunc_fview (G : PNet) {s t : ℕ} (h : s ≤ t) (v : ℕ) : (fview G v t).trunc s = fview G v s := by induction s
