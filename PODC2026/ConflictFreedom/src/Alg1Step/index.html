:= ⟨C.mem (.A r), C.mem (.B r), fun i => locAt (C.loc i) r⟩ variable [Inhabited A.Resp] [Inhabited A.Op] [NeZero n] /-- Replace the local state of `p`. -/ def setLoc (C : Cfg A n) (p : Fin n) (l : Loc A n) : Cfg A n := { C with loc := Function.update C.loc p l } /-- Replace the state of base object `o` and the local state of `p`. -/ def setMemLoc (C : Cfg A n) (o : Obj n) (v : (base A n o).State) (p : Fin n) (l : Loc A n) : Cfg A n := { C with mem := Function.update C.mem o v, loc := Function.update C.loc p l } /-- The local state after the last read of `S` (line 5) in phase `readS j best`. -/ noncomputable def Loc.afterRead (l : Loc A n) (b : ℕ × Tr A n) : Loc A n := l.startRound b.1 b.2 /-- The explicit forms of an internal step of process `p` from `C` to `C'`. -/ inductive TauStep (C : Cfg A n) (p : Fin n) : Cfg A n → Prop | readS (j : ℕ) (best : ℕ × Tr A n) (hj : j < n) (hph : (C.loc p).phase = .readS j best) (hlt : j + 1 < n) : TauStep C p (setLoc C p { C.loc p with phase := .readS (j + 1) (better best (C.mem (.S ⟨j, hj⟩))) }) | readLast (j : ℕ) (best : ℕ × Tr A n) (hj : j < n) (hph : (C.loc p).phase = .readS j best) (hlt : ¬ j + 1 < n) : TauStep C p (setLoc C p ((C.loc p).afterRead (better best (C.mem (.S ⟨j, hj⟩))))) | gca0 (g : GCA.Loc n (Tr A n)) (hph : (C.loc p).phase = .gca g) (hpc : g.pc = 0) : TauStep C p (setMemLoc C (.A (C.loc p).r) (Function.update (C.mem (.A (C.loc p).r)) p (some g.inp)) p { C.loc p with phase := .gca { g with pc := 1 } }) | gca1 (g : GCA.Loc n (Tr A n)) (hph : (C.loc p).phase = .gca g) (hpc : g.pc = 1) : TauStep C p (setLoc C p { C.loc p with phase := .gca { g with pc := 2, av := C.mem (.A (C.loc p).r) } }) | gca2 (g : GCA.Loc n (Tr A n)) (hph : (C.loc p).phase = .gca g) (hpc : g.pc = 2) : TauStep C p (setMemLoc C (.B (C.loc p).r) (Function.update (C.mem (.B (C.loc p).r)) p (some (GCA.candidate g.av, GCA.compatFlag g.av))) p { C.loc p with phase := .gca { g with pc := 3 } }) | gca3 (g : GCA.Loc n (Tr A n)) (hph : (C.loc p).phase = .gca g) (hpc : g.pc ≠ 0 ∧ g.pc ≠ 1 ∧ g.pc ≠ 2) : TauStep C p (setLoc C p ((C.loc p).afterGCA { g with pc := 4, bv := C.mem (.B (C.loc p).r) })) | writeS (hph : (C.loc p).phase = .writeS) : TauStep C p (setMemLoc C (.S p) ((C.loc p).r, (C.loc p).s) p { C.loc p with phase := .done }) /-- Every internal step has one of the explicit forms of `TauStep`. -/ theorem step_tau_cases {C C' : Cfg
