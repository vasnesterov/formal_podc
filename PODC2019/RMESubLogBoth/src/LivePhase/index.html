unfold LI at hLp; rw [h1] at hLp obtain ⟨-, -, ⟨m, rfl⟩, -⟩ := hLp rcases step_l25 h' h1 with ⟨-, h2⟩ | ⟨-, rfl⟩ · exact .inl ⟨hnode ▸ hn, .inr (.inr (.inr h2))⟩ · exact .inl ⟨hn, .inr (.inr (.inl ⟨_, h1⟩))⟩ · exact .inr (step_l26 h' h1) · have h1 := Next.pc_other h he hcr have h2 := Next.node_other h he exact .inl ⟨h2 ▸ hn, h1 ▸ hpc⟩ /-- **Phase 2.** After the last crash, once nobody runs the recovery code or `RLock`, a process at lines 14–26 eventually enters the CS. -/ theorem phase2_cs (r : Run k) (hF : Fair r) {T : ℕ} (hT : CrashFreeFrom r T) (hNR : ∀ t, T ≤ t → NoRec (r.cfg t)) {p : Fin k} {yp : ℕ} {t0 : ℕ} (ht0 : T ≤ t0) (hph : Ph2 (r.cfg t0) p yp) : ∃ t, t0 ≤ t ∧ (r.cfg t).InCS p := by let Goal : ℕ → Prop := fun t => ∃ t', t0 ≤ t' ∧ t' ≤ t ∧ (r.cfg t').InCS p have hph' : ∀ t, t0 ≤ t → ¬ Goal t → Ph2 (r.cfg t) p yp := by intro t ht induction t, ht using Nat.le_induction with | base => intro _; exact hph | succ t ht ih => intro hg have hg' : ¬ Goal t := fun ⟨t', h1, h2, h3⟩ => hg ⟨t', h1, by omega, h3⟩ rcases ph2_next (r.reach t) (r.next t) (ih hg') (hT t (by omega) p) with h1 | h1 · exact h1 · exact absurd ⟨t + 1, by omega, le_rfl, ⟨yp, h1⟩⟩ hg have hHO : ∀ t, t0 ≤ t → ¬ Goal t → ∃ s, HO (r.cfg t) s := by intro t ht hg obtain ⟨own, hG, hL⟩ := inv_reach (r.reach t) obtain ⟨s, x, hs, hx, -⟩ := (hph' t ht hg).holder hG hL (hNR t (by omega)) exact ⟨s, x, hs, hx⟩ let d : ℕ → Fin k := fun t => if h : ∃ s, HO (r.cfg t) s then h.choose else p have hdHO : ∀ t, t0 ≤ t → ¬ Goal t → HO (r.cfg t) (d t) := by intro t ht hg have h := hHO t ht hg simp only [d, dif_pos h]; exact h.choose_spec have hstep := fun t (ht : t0 ≤ t) (hg : ¬ Goal t) => token_step (inv_reach (r.reach t)) (inv_reach (r.reach (t + 1))) (hNR t (by omega)) (s := match r.ev t with | .step s => s | .crash s => s) (by have := r.next t cases he : r.ev t with | step s => rw [he] at this;
