import Papers.PODC2019.HardnessOptimization.MDSGraph /-! # The lower-bound graphs for weighted max-cut (Section 2.4.1) Bachrach, Censor-Hillel, Dory, Efron, Leitersdorf, Paz. *Hardness of Distributed Optimization.* PODC 2019 (arXiv:1905.10284), Section 2.4.1. `k = 2 ^ L`, a row index `j ∈ [k]` is a bit string `j : Idx L = Fin L → Bool`. Vertices (`CV L`), with `s = false` for the `A` side and `s = true` for the `B` side, and `z = false` for the first row (`A₁`, `B₁`), `z = true` for the second (`A₂`, `B₂`): * `row s z j`: `a^j_1, a^j_2, b^j_1, b^j_2`; * `gad s z h t`: `t^h_S` (`t = true`) and `f^h_S` (`t = false`) for `S = A_z / B_z`; * `cc s`: `C_A`, `C_B`; `nn s`: `N_A`, `N_B`; `cbar`: `C̄_A`. Edges (`CE L`, each undirected edge listed once, from `src` to `dst`), with weights `wt x y`: * `cn s`: `C_s – N_s`, `cbar s`: `C_s – C̄_A`, weight `k⁴`; * `cyc s z h t`: the 4-cycle `t_{A_z}^h – f_{A_z}^h – t_{B_z}^h – f_{B_z}^h – t_{A_z}^h`, weight `k⁴` (`cyc s z h true = t_S – f_S`, `cyc s z h false = f_S – t_{S'}`, `S'` the other side); * `bin s z j h`: `row s z j – (t^h if j_h = 1 else f^h)` (i.e. `Bin(s^j)`), weight `2k²`; * `rc s z j`: `row – C_s`, weight `2k² log k − k²`; * `rn s z j`: `row – N_s`, weight `Σ_j x_{i,j}` (first row) resp. `Σ_j x_{j,i}` (second row) (`y` on the `B` side); * `inp s i j`: `row s false i – row s true j`, weight `1` if `x_{i,j} = 0` (resp. `y`), else `0`. The weighted graph is the symmetric weight function `W x y : CV L → CV L → ℕ` (`0` = no edge). Edges of weight `0` (input edges with `x_{i,j} = 1`, and `N`-edges with zero row sum) are thus absent; this is immaterial for cut weights. -/ namespace HardnessOptimization namespace MaxCut /-- Vertices of the max-cut lower-bound graph. -/ inductive CV (L : ℕ) where | row (s z : Bool) (j : Idx L) | gad (s z : Bool) (h : Fin L) (t : Bool) | cc (s : Bool) | nn (s : Bool) | cbar deriving : Bool) (j : Idx L) (h : Fin L) | rc (s z : Bool) (j : Idx L) | rn (s z : Bool) (j : Idx L) | inp (s : Bool) (i j : Idx L) deriving DecidableEq, Fintype variable {L : ℕ} namespace CE /-- First => .row s false i /-- Second endpoint of an edge. -/ def dst : CE L → CV L | cn s => .nn s | cbar _ => .cbar | cyc s z h true => .gad s z h false | cyc s z h false => .gad (!s) z h true | bin s z j h [src, dst] | _ => simp [src, dst] end CE /-- Row sum `Σ_j m_{i,j}`. -/ def rsum (m : Idx L → Idx L → Bool) (i : Idx L) : ℕ := ∑ j, (m i j).toNat /-- Column sum `Σ_i m_{i,j}`. -/ def csum (m : Idx L →
