import Papers.PODC2019.AnonymousMutex.Alg1 /-! # Algorithm 1 is symmetric and uses only reads, writes and snapshots Aghazadeh, Imbs, Raynal, Taubenfeld, Woelfel, PODC 2019 (arXiv:1810.03723), Section 3. Identifier renamings `σ : Equiv.Perm Id` act on local states by renaming the own identifier and the entries of the last snapshot. Algorithm 1 is equivariant under this action (`alg1_symmetric`: *symmetry with equality*, identifiers are only stored and compared), and every read-modify-write it performs is a plain read or a plain write (`alg1_readWrite`). -/ set_option linter.unusedSectionVars false open Cslib.SharedMemory namespace AnonymousMutex.Alg1 variable {Id : Type} {m : ℕ} /-- Identifier renamings act on local states (own identifier and snapshot entries). -/ instance : MulAction (Equiv.Perm Id) (St Id m) where smul σ s := ⟨σ s.me, s.pc, fun x => (s.view x).map σ⟩ one_smul s := by cases s change St.mk _ _ _ = _ simp mul_smul σ τ s := by change St.mk _ _ _ = St.mk _ _ _ congr 1 funext x change _ = Option.map σ (Option.map τ (s.view x)) cases s.view x <;> rfl @[simp] lemma smul_me (σ : Equiv.Perm Id) (s : St Id m) : (σ • s).me = σ s.me := rfl @[simp] lemma smul_pc (σ : Equiv.Perm Id) (s : St Id m) : (σ • s).pc = s.pc := rfl @[simp] lemma smul_view (σ : Equiv.Perm Id) (s : St Id m) (x : Fin m) : (σ • s).view x = (s.view x).map σ := rfl lemma smul_with_pc (σ : Equiv.Perm Id) (s : St Id m) (pc : PC m) : ({ σ • s with pc := pc } : St Id m) = σ • { s with pc := pc } := rfl lemma map_eq_some_smul_iff (σ : Equiv.Perm Id) (v : Option Id) (a : Id) : v.map σ = some (σ a) ↔ v = some a := by cases v <;> simp variable [DecidableEq Id] lemma nextOwn_smul (σ : Equiv.Perm Id) (view : Fin m → Option Id) (me : Id) (i : ℕ) : nextOwn (fun x => (view x).map σ) (σ me) i = nextOwn view me i := by simp only [nextOwn, map_eq_some_smul_iff] lemma owned_smul (σ : Equiv.Perm Id) (view : Fin m → Option Id) (me : Id) : owned (fun x => (view x).map σ) (σ me) = owned view me := by simp only [owned, map_eq_some_smul_iff] lemma cnt_smul (σ : Equiv.Perm Id) (view : Fin m → Option Id) : cnt (fun x => (view x).map σ) = cnt view := by unfold cnt have : (Finset.univ.image fun x => (view x).map σ) = (Finset.univ.image view).image (Option.map σ) := by rw [Finset.image_image];
