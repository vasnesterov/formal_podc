`sent` is the set of messages sent so far. -/ inductive Trans (V : Variant D) (QS : Finset (Finset P)) (sent : Multiset (P × P × Msg P D)) (p : P) (l₀ : Loc P D) : Loc P D → Multiset (P × Msg P D) → Option (Label P D.spec) → Prop | inv {o l out} : invokeStep p o l₀ = some (l, out) → Trans V QS sent p l₀ l out (some (.inv p o)) | internal {a l out r} : internalStep V QS p a l₀ = some (l, out, r) → Trans V QS sent p l₀ l out (r.map (Label.ret p)) | recv {src m} : (src, p, m) ∈ sent → Trans V QS sent p l₀ (receiveStep V p src m l₀).1 (receiveStep V p src m l₀).2 none /-- Classification of the steps of Algorithm 2. -/ lemma step_cases {C C' : Config P (Loc P D) (Msg P D)} {st : Step P D.spec (Msg P D) Act} {ℓ : Option (Label P D.spec)} (hnet : ∀ x ∈ C.net, x ∈ C.sent) (h : (alg V QS).step C st = some (C', ℓ)) : (C'.loc = C.loc ∧ C'.sent = C.sent ∧ ℓ = none ∧ ∀ x ∈ C'.net, x ∈ C.net) ∨ ∃ p l out, Trans V QS C.sent p (C.loc p) l out ℓ ∧ C'.loc = Function.update C.loc p l ∧ C'.sent = C.sent + outMsgs p out ∧ ∀ x ∈ C'.net, x ∈ C.net ∨ x ∈ outMsgs p out := by cases st with | inv p o => simp only [Protocol.step, alg] at h split_ifs at h obtain ⟨⟨l, out⟩, h1, hx⟩ := Option.map_eq_some_iff.1 h simp only [Prod.mk.injEq] at hx;
