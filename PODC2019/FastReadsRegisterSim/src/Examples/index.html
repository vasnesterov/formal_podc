import Papers.PODC2019.FastReadsRegisterSim.Theorem41 import Cslib.Computability.Distributed.SharedMemory.Atomic /-! # Non-vacuity: the trivial simulation of a binary register For `k = 2` (`m = 1`) a binary register is simulated by one binary register (CSLib's `atomicImpl`): every operation applies itself to the base register in one step. It is wait-free, atomic (hence regular and safe), and every read takes one step. So * the hypotheses of Theorems 3.1/3.2 and Corollary 3.5 are satisfiable (`trivialSim_hyps`); * the definitions `Sim.Atomic` and `Sim.ReadStepsLE` are not contradictory, and the hypothesis `k > 2` of Theorem 4.1 is necessary (`theorem_4_1_needs_k_gt_two`). -/ set_option linter.unusedSectionVars false namespace Podc.FastReadsRegisterSim open Cslib.SharedMemory variable {P : Type} [DecidableEq P] variable (P) in /-- The trivial simulation of a binary register (initial value `false`) by one binary register. -/ def trivialSim : Sim Bool false Unit P (AtomicLoc (register Bool false)) := atomicImpl (register Bool false) false P lemma trivialSim_atomic (w : P) : (trivialSim P).Atomic w := by intro α hα _ obtain ⟨h, hh⟩ := Atomic.exists_linFun (A := register Bool false) (P := P) (r₀ := false) obtain ⟨C, hC⟩ := hα exact ⟨h α, (hh α C hC).lin⟩ lemma trivialSim_waitFree : (trivialSim P).WaitFree := Atomic.waitFree /-- Remaining steps of an operation of the trivial simulation. -/ def phi : AtomicLoc (register Bool false) → ℕ | .apply _ => 1 | _ => 0 lemma numSteps_accessOf_ne (D : Config (register Bool false) (BinRegs Unit) P (AtomicLoc (register Bool false))) {ℓ : Label P (register Bool false)} {r : P} (hℓ : ℓ.proc ≠ r) : numSteps r ((trivialSim P).accessOf D ℓ) = 0 := by cases ℓ with | tau q => simp only [Label.proc] at hℓ simp only [Sim.accessOf] split <;> simp [numSteps, hℓ] | inv q o => simp [Sim.accessOf, numSteps] | ret q v => simp [Sim.accessOf, numSteps] lemma numSteps_le_phi (r : P) : ∀ (α : List (Label P (register Bool false))) (D : Config (register Bool false) (BinRegs Unit) P (AtomicLoc (register Bool false))), D.pend r ≠ none → (∀ v, Label.ret r v ∉ α) → ((trivialSim P).runFrom D α).isSome → numSteps r ((trivialSim P).accesses D α) ≤ phi (D.loc r) := by intro α induction α with | nil => intro D _ _ _;
