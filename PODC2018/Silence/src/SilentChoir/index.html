import Papers.PODC2018.Silence.Defs /-! # The Silent Choir Theorem (Theorem 2 of Goren–Moses) If `K_i(v_j = 1)` holds at time `m` of a run `r` of any protocol in `γ_f`, and `r` contains no message chain from `j` to `i`, then `m > 0` and `|F_r ∪ S_j^r(m − 1)| > f`. Proof (as in the paper): otherwise build the run `r'` (`choirRun`) in which `v_j = 0` and every process `h ∈ S_j^r(m − 1)` crashes, without sending anything, in the round right after the one in which it first got a chain from `j` (if it has not crashed earlier anyway in `r`). Processes not yet reached by a chain behave identically in both runs (`hist_choirRun`), so `r ≈ᵢᵐ r'`. -/ namespace Papers.PODC2018.Silence open Cslib.SyncCrash Classical variable {P M O : Type*} [Fintype P] {alg : Cslib.Byzantine.LockStep.Protocol P Bool M O} section Construction variable (alg) (r : Run P Bool) (j : P) (m : ℕ) /-- The first time at which a chain from `(j, 0)` reaches `z` (`0` if never). -/ noncomputable def kk (z : P) : ℕ := if h : ∃ t, Chain alg r j 0 z t then Nat.find h else 0 /-- The run `r'` of the proof of Theorem 2. -/ noncomputable def choirRun : Run P Bool where input := Function.update r.input j false crash z := if Chain alg r j 0 z (m - 1) then some (match r.crash z with | some c => min c (kk alg r j z + 1) | none => kk alg r j z + 1) else r.crash z reach z := if Chain alg r j 0 z (m - 1) ∧ ¬ (∃ c, r.crash z = some c ∧ c ≤ kk alg r j z) then ∅ else r.reach z end Construction variable {r : Run P Bool} {j : P} {m : ℕ} lemma chain_kk {z : P} {t : ℕ} (h : Chain alg r j 0 z t) : Chain alg r j 0 z (kk alg r j z) ∧ kk alg r j z ≤ t := by have he : ∃ t, Chain alg r j 0 z t := ⟨t, h⟩ simp only [kk, dif_pos he] exact ⟨Nat.find_spec he, Nat.find_min' he h⟩ lemma lt_kk {z : P} {t t' : ℕ} (h : Chain alg r j 0 z t') (hn : ¬ Chain alg r j 0 z t) : t < kk alg r j z := by by_contra hle exact hn ((chain_kk h).1.trans_le (by omega)) /-- Processes not reached by time `ℓ` are alive at `ℓ` in `r'` iff they are in `r`. -/ lemma choirRun_alive_iff {z : P} {ℓ : ℕ} (hz : ¬ Chain alg r j 0 z ℓ) : (choirRun alg r j m).Alive z ℓ ↔ r.Alive z ℓ := by simp only [Run.Alive, choirRun] split_ifs with hS · have hk := lt_kk hS hz cases hc : r.crash z with | none => simp;
