import Papers.PODC2013.ConstantBarriersMutex.Progress import Papers.PODC2013.ConstantBarriersMutex.Stuck /-! # The waiting section (statements 23–27) guarantees an exit after the barrier The key use of statement 25 (`await exits − e ≥ 2 ∨ lock ∈ {p, −1}`): a process that completes its barrier (statement 22) at time `ta` and never enters the critical section afterwards forces some process to enter the critical section after `ta` (`Run.waiting_entry`). Without the `exits − e ≥ 2` disjunct the process could wait at statement 27 although the lock owner read the tree before `ta`. -/ namespace Podc.ConstantBarriersMutex variable {h : ℕ} namespace Run variable (r : Run h) {p : Proc h} /-- Program counters of a process that has completed its barrier at `ta` while no process entered the critical section since `ta` and the lock is never handed to it. -/ def WaitOK (ta t : ℕ) (p : Proc h) : Prop := match (r.s t).pc p with | .e23 | .e24 => True | .e25a e | .e25b e => (r.s ta).ent ≤ e + 1 | .e26 => (r.s t).mem.lock = none | _ => False /-- Measure for the waiting section when the lock is free. -/ def waitμ : PC h → ℕ | .e23 => 1 | .e24 => 4 | .e25a _ => 3 | .e25b _ => 2 | .e26 => 1 | _ => 0 lemma waitOK_buf {ta t : ℕ} (hw : r.WaitOK ta t p) : (r.s t).buf p = [] := by have hL := (r.inv t).loc p unfold WaitOK at hw; unfold LI at hL cases hpc : (r.s t).pc p <;> simp only [hpc] at hw hL <;> first | exact hL.1 | exact hw.elim lemma waitOK_cases {ta t : ℕ} (hw : r.WaitOK ta t p) : (r.s t).pc p = .e23 ∨ (r.s t).pc p = .e24 ∨ (∃ e, (r.s t).pc p = .e25a e ∧ (r.s ta).ent ≤ e + 1) ∨ (∃ e, (r.s t).pc p = .e25b e ∧ (r.s ta).ent ≤ e + 1) ∨ ((r.s t).pc p = .e26 ∧ (r.s t).mem.lock = none) := by unfold WaitOK at hw cases hpc : (r.s t).pc p <;> simp only [hpc] at hw <;> first | exact hw.elim | simp [hw] /-- **Statement 25 at work.** A process that completes its barrier at `ta` and never enters the critical section afterwards: some process enters the critical section after `ta`. -/ theorem waiting_entry (hf : r.Fair) {ta : ℕ} (hpc : (r.s ta).pc p = .e23) (hno : ∀ t ≥ ta, (r.s t).pc p ≠ .cs) : ∃ t ≥ ta, (r.s (t + 1)).ent = (r.s t).ent + 1 := by by_contra hG push_neg at hG have hG' : ∀ t ≥ ta, (r.s (t + 1)).ent = (r.s t).ent := fun t ht => by rcases r.ent_succ t with h1 | h1 · exact h1 · exact absurd h1 (hG t ht) have hentc : ∀ t ≥ ta, (r.s t).ent = (r.s ta).ent := by intro t ht induction t, ht using Nat.le_induction with | base => rfl | succ t ht ih => rw [hG' t ht, ih] have hE : ∀ t ≥ ta, ((r.s t).pc p).Entry := r.entry_forever (by rw [hpc]; simp [PC.Entry]) hno -- the lock is never handed to `p` have hnl : ∀ t ≥ ta, (r.s t).mem.lock ≠ some p := by intro t ht hl obtain ⟨t', ht', hcs⟩ := r.handed_enters hf hl (hE t ht) exact hno t' (by omega) hcs -- the invariant of the waiting section have hok : ∀ t ≥ ta, r.WaitOK ta t p := by intro t ht induction t, ht using Nat.le_induction with | base => unfold WaitOK;
