import Cslib.Computability.Distributed.Local.Locality import Podc.Foundations.Combinatorics.SimpleGraph.StraightLineDrawing /-! # The triangulated grid, identifier assignments and local views The graph `G` of Hilke–Lenzen–Suomela (Figure 1a) is a `W × H` piece of the triangular lattice: node `(x, y)` (`x < W` columns, `y < H` rows) is adjacent to `(x ± 1, y)`, `(x, y ± 1)` and `(x + 1, y + 1)`, `(x - 1, y - 1)` (`grid W H`, the triangular lattice pulled back along the integer embedding `pt`). It is planar (`grid_isStraightLinePlanar`). In the LOCAL model with unique identifiers `{0, …, n - 1}` (`n = W · H`), an instance is an identifier assignment `σ : GV W H ≃ Fin n`; the network is the copy `lab σ` of the grid whose nodes are the identifiers. `view T σ v` is the `(2T+1) × (2T+1)` array of identifiers around `v` (`none` outside the grid). The key fact (`out_eq_of_view_eq`): a `T`-round LOCAL algorithm gives the same output at two nodes whose views agree (`Cslib.Local.IsLocal`), because the radius-`T` neighbourhood of a node lies in this array and the triangular lattice is translation invariant. This is the paper's "the local output of `v` only depends on the identifiers in the `k × k` region `R_v`" (Figure 1c). -/ namespace Podc.PlanarMDSLowerBound open SimpleGraph Cslib.SimpleGraph /-- Grid nodes: `x < W` (column), `y < H` (row). -/ abbrev GV (W H : ℕ) := Fin W × Fin H /-- Integer coordinates of a grid node. -/ def pt {W H : ℕ} (p : GV W H) : ℤ × ℤ := ((p.1 : ℕ), (p.2 : ℕ)) lemma pt_injective {W H : ℕ} : Function.Injective (pt (W := W) (H := H)) := by intro p q h simp only [pt, Prod.mk.injEq, Nat.cast_inj] at h exact Prod.ext (Fin.ext h.1) (Fin.ext h.2) /-- The `W × H` triangulated grid (Figure 1a of the paper, as one strip of `r` blocks). -/ def grid (W H : ℕ) : SimpleGraph (GV W H) := triangularLattice.comap pt lemma grid_adj {W H : ℕ} {p q : GV W H} : (grid W H).Adj p q ↔ TriAdj (pt p) (pt q) := Iff.rfl /-- The triangulated grid is planar (it has a straight-line planar drawing). -/ theorem grid_isStraightLinePlanar (W H : ℕ) : IsStraightLinePlanar (grid W H) := isStraightLinePlanar_comap_triangularLattice pt_injective /-- The network with identifier assignment `σ`: the copy of the grid on the identifiers. -/ def lab {W H n : ℕ} (σ : GV W H ≃ Fin n) : SimpleGraph (Fin n) := (grid W H).comap σ.symm lemma lab_adj {W H n : ℕ} {σ : GV W H ≃ Fin n} {a b : Fin n} : (lab σ).Adj a b ↔ TriAdj (pt (σ.symm a)) (pt (σ.symm b)) := Iff.rfl /-- `lab σ` is isomorphic to the grid. -/ def labIso {W H n : ℕ} (σ : GV W H ≃ Fin n) : lab σ ≃g grid W H := ⟨σ.symm, Iff.rfl⟩ /-! ### Distances -/ /-- A function changing by at most `1` along edges changes by at most the length of a walk. -/ lemma abs_sub_le_length {V : Type*} {G : SimpleGraph V} (f : V → ℤ) (hf : ∀ u v, G.Adj u v → |f u -
