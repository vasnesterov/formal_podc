import Podc.Computability.Distributed.SharedMemory.Awareness import Cslib.Computability.Distributed.SharedMemory.Objects.MaxRegister import Mathlib.Tactic.IntervalCases /-! # The paper's Definitions 1–5 taken literally, and counterexamples to Lemma 2 and Claim 1 Hendler–Khait (PODC'14) define (Definition 1) an event `e` of process `p` on object `o` in `E` to be *invisible* if it does not change the value of `o`, or if it is followed by a `write` `e'` to `o` such that no event between `e` and `e'` is applied to `o` and `p` takes no step in between. Definition 2 makes a read/CAS event *aware* of a write/CAS event `e_q` on the same object that is visible in a prefix containing both, and closes awareness under chaining; Definition 3 makes `p` aware of `q` if some event of `p` is aware of some event of `q`; Definition 5 calls `p` *hidden* if no other process is aware of `p`. Here the definitions are transcribed on finite event sequences (`Ev`) and evaluated on a concrete execution of a single-writer max register (`toyImpl`): the writer `q` writes `5` twice in a row (the second write is trivial and overwrites the first), then the reader `r` reads `5`. Neither write is visible, so nobody is aware of `q`: `q` is hidden, and no event is aware of an event of `q`. Yet removing `q`'s events does not leave an execution (the reader would read `0`). * `lemma2_false` — Lemma 2 is false as stated (with `p := q`, nothing is aware of `q`); * `claim1_false` — Claim 1 is false as stated (`{q}` is a set of hidden processes). (For Lemma 2 we even use a *more generous* closure than the paper's Definition 2, which in its transitive clause also requires `e_p` to access `o`; fewer awareness pairs would only make the counterexample easier.) The flaw is the combination of the two invisibility clauses of Definition 1: an overwritten write is invisible, and so is the trivial write that overwrites it. `Cslib.SharedMemory.Awareness` replaces these definitions by a sound data-flow semantics. -/ namespace Podc.ReadUpdateTradeoffs.PaperDefs open Cslib.SharedMemory Cslib.SharedMemory.Awareness /-! ### Events and the paper's definitions -/ /-- An event: process, base object, primitive. -/ structure Ev (P ι V : Type) where proc : P obj : ι prim : CASOp V variable {P ι V : Type} [DecidableEq ι] [DecidableEq V] /-- The memory before the `n`-th event of `E`, starting from `m₀`. -/ def memBefore (v₀ : V) (m₀ : ι → V) (E : List (Ev P ι V)) (n : ℕ) : ι → V := (E.take n).foldl (fun m e => Function.update m e.obj ((casObject V v₀).δ (m e.obj) e.prim).1) m₀ /-- The `n`-th event changes the value of its object. -/ def Changes (v₀ : V) (m₀ : ι → V) (E : List (Ev P ι V)) (n : ℕ) : Prop := ∃ e, E[n]? = some e ∧ ((casObject V v₀).δ (memBefore v₀ m₀ E n e.obj) e.prim).1 ≠ memBefore v₀ m₀ E n e.obj /-- `e` is a write. -/ def IsWrite (e : Ev P ι V) : Prop := ∃ v, e.prim = .write v /-- `e` is a read. -/ def IsRead (e : Ev P ι V) : Prop := e.prim = .read /-- `e` is a CAS. -/ def IsCAS (e : Ev P ι V) : Prop := ∃ a b, e.prim = .cas a b /-- **Definition 1** (invisible event). -/ def Invisible (v₀ : V) (m₀ : ι → V) (E : List (Ev P ι V)) (n : ℕ) : Prop := ¬ Changes v₀ m₀ E n ∨ ∃ e, E[n]? = some e ∧ ∃ m e', n < m ∧ E[m]? = some e' ∧ IsWrite e' ∧ e'.obj = e.obj ∧ ∀ k f, n < k → k < m → E[k]? = some f → f.proc ≠ e.proc ∧ f.obj ≠ e.obj /-- Visible = not invisible. -/ def Visible (v₀ : V) (m₀ : ι → V) (E : List (Ev P ι V)) (n : ℕ) : Prop := n < E.length ∧ ¬ Invisible v₀ m₀ E n /-- **Definition 2** (event `a` is aware of event `b`), with a generous transitive closure. -/ inductive Aware (v₀ : V) (m₀ : ι → V) (E : List (Ev P ι V)) : ℕ → ℕ → Prop | base {a b : ℕ} {ea eb : Ev P ι V} : E[a]? = some ea → E[b]? = some eb → (IsWrite eb ∨ IsCAS eb) → ea.obj = eb.obj → b < a → (IsRead ea ∨ IsCAS ea) → (∃ m, a < m ∧ m ≤ E.length ∧ Visible v₀ m₀ (E.take m) b) → Aware v₀ m₀ E a b | trans {a b c : ℕ} : Aware v₀ m₀ E c b → Aware v₀ m₀ E a c → Aware v₀ m₀ E a b /-- **Definition 3** (process `p` is aware of process `q`). -/ def AwareOf (v₀ : V) (m₀ : ι → V) (E : List (Ev P ι V)) (p q : P) : Prop := p = q ∨ ∃ a b ea eb, E[a]? = some ea ∧ E[b]? = some eb ∧ ea.proc = p ∧ eb.proc = q ∧ Aware v₀ m₀ E a b /-- **Definition 5** (hidden process). -/ def HiddenProc (v₀ : V) (m₀ : ι → V) (E : List (Ev P ι V)) (p : P) : Prop := ∀ p', AwareOf v₀ m₀ E p' p → p = p' /-! ### The trace of an execution -/ variable {A : ObjSpec} {L : Type} [DecidableEq P] /-- The base-object events of an execution, in order. -/ noncomputable def traceFrom (I : Impl A (RWC ι V v₀) P L) : Config A (RWC ι V v₀) P L → List (Label P A) → List (Ev P ι V) | _, [] => [] | C, ℓ :: α => match I.step C ℓ with | none => [] | some C' => (match ℓ with | .tau p => match I.act p (C.loc p) with | .access i o _ => [⟨p, i, o⟩] | .ret _ _ => [] | _ => []) ++ traceFrom I C' α /-! ### A single-writer max register -/ /-- Local states of the toy implementation. -/ inductive Loc where | idle (mx : ℕ) | w₁ (mx : ℕ) | w₂ (mx : ℕ) | done (mx : ℕ) | rd (mx : ℕ) | rdone (mx v : ℕ) deriving DecidableEq /-- The maximum written so far by this process. -/ def Loc.mx : Loc → ℕ | .idle m | .w₁ m | .w₂ m | .done m | .rd m | .rdone m _ => m /-- A single-writer max register on one CAS object (initially `0`): process `0` is the only writer; `WriteMax(v)` writes `max(previous, v)` **twice**; `ReadMax` reads the register. -/ def toyImpl : Impl maxRegObj (RWC Unit ℕ 0) (Fin 2) Loc where memInit := fun _ => 0 locInit := fun _ => .idle 0 invoke := fun _ l o => match o with | .tau 1, .ret 1 5] /-- Its events. -/ def evE : List (Ev (Fin 2) Unit ℕ) := [⟨0, (), .write 5⟩, ⟨0, (), .write 5⟩, ⟨1, (), .read⟩] lemma execE_valid : toyImpl.Valid execE := Impl.valid_of_crun toyImpl
