import Papers.PODC2020.RecoverableMutexRMR.Model /-! # The invariant of the Chan–Woelfel algorithm (definitions) `Inv s g` is an inductive invariant of configurations `s` with ghost state `g`. It contains the paper's Lemma 2.1 (in state form), the potential `Φ = D + |I| − C ≥ 0` of Lemma 2.3, and a second potential `Ψ ≥ Φ` bounding the number of skipped tickets by twice the number of crashes (Lemma 2.10 / the claim "the total number of skipped tickets is at most twice the number of process crashes" of Section 3.1). -/ namespace Podc.RecoverableMutexRMR variable {P : Type} [DecidableEq P] [Fintype P] /-- `p ∈ I_t` (Section 3.3): `p` holds a ticket that is not skipped and has not incremented `D` since obtaining it. -/ def inI (s : Cfg P) (g : Ghost P) (p : P) : Prop := g.cur p ≠ 0 ∧ s.A (g.cur p) ≠ .skipped ∧ g.pend p = true instance (s : Cfg P) (g : Ghost P) : DecidablePred (inI s g) := fun p => by unfold inI; infer_instance /-- `|I_t|`. -/ def Isize (s : Cfg P) (g : Ghost P) : ℕ := (Finset.univ.filter (inI s g)).card /-- The potential `Φ_t = D_t + |I_t| − C_t` of Lemmas 2.2–2.3. -/ def Phi (s : Cfg P) (g : Ghost P) : ℤ := (s.D : ℤ) + Isize s g - s.C /-- Processes holding a "crash credit" for a ticket that is not skipped. -/ def crCount (s : Cfg P) (g : Ghost P) : ℕ := (Finset.univ.filter fun p => g.cr p = true ∧ s.A (g.cur p) ≠ .skipped).card /-- Lost tickets that are still unclaimed and unskipped. -/ def lostCount (s : Cfg P) (g : Ghost P) : ℕ := (g.lost.filter fun x => s.A x = .empty).card /-- Number of processes about to execute line 15. -/ def nR15 (s : Cfg P) : ℕ := (Finset.univ.filter fun p => s.pc p = .r15).card /-- Number of skipped tickets (entries `A[x] = −1`). -/ def nSkipped (s : Cfg P) : ℕ := ((Finset.range (s.C + 1)).filter fun x => s.A x = .skipped).card /-- The second potential `Ψ = Φ + crCount + lostCount`. -/ def Psi (s : Cfg P) (g : Ghost P) : ℤ := Phi s g + crCount s g + lostCount s g /-- `q` is in its `Exit()` section after line 10, on its way to move `Token` past its own ticket `T + 1`. -/ def OnTrack (s : Cfg P) (q : P) : Prop := match s.pc q with | .x9 | .x10 | .r15 | .r16 | .m34 .ex => True | .m35 .ex ℓ | .m36 .ex ℓ _ | .m38 .ex ℓ _ _ | .m42 .ex ℓ => ℓ = s.T | .m40 .ex ℓ q' | .m41 .ex ℓ q' => ℓ = s.T ∧ q' = q | _ => False /-- Invariant of the return address of a `Move_Token()` call. -/ def RetInv (s : P) : Prop := ∀ x, s.A x = .claimed q → s.T < x → (s.W q).1 = x ∨ (x = s.T + 1 ∧ s.W q = (0, false) ∧ s.X q = true ∧ OnTrack s q) /-- The inductive invariant. -/ structure Inv (s : Cfg P) (g : Ghost (0, false) pend_pc : ∀ p, g.pend p = true → PendPC (s.pc p) ∨ s.A (g.cur p) = .skipped cr_pc : ∀ p, CrPC (s.pc p) → g.pend p = true ∨ g.cr p = true l3_cr : ∀ p, s.pc p = .l3 → g.pend p = true ∨ g.cr p = true ∨ s.W p = (0, false) I_cr : ∀ p, inI s g p → g.cr p = false lost_wit : ∀ x, x ≠ 0 → x ≤ s.C → s.A x = .empty → x ∈ g.lost ∨ ∃ o, g.own x = some o ∧ g.cur o = x ∧ (g.pend o = true ∨ g.cr o = true) /-- Lemma 2.3. -/ phi_nonneg : 0 ≤ Phi s g psi_bound : Psi s g + 2 * (nR15 s : ℤ) + nSkipped s ≤ 2 * (g.ncr : ℤ) loc : ∀ p, LI s g p end Podc.RecoverableMutexRMR
