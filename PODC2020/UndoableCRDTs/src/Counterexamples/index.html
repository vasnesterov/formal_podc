import Papers.PODC2020.UndoableCRDTs.Main /-! # Sharpness remarks (Dolan, PODC 2020) 1. **Axiom 1 must include `p = q`.** The paper applies Commutativity with `p = q` ("any valid action can be done twice", Fact 2). This is essential: the nonnegative counter `nnCounter` (states `ℕ`, `dec` only applies in nonzero states) satisfies Axiom 1 for all *distinct* `p ≠ q`, Axiom 2, finiteness and reachability, but it is not equivalent to *any* undoable CRDT (in particular, to no tuple of counters): `nnCounter_not_equiv`. Under the paper's Axiom 1 it is not a CRDT at all (`nnCounter_not_commutative`). 2. **Dead operations.** The definitions allow a primitive operation that applies in no state. Section 3 says "all operations are always valid"; this holds only for operations that apply in *some* state (`all_operations_valid`). `deadOp` is an undoable CRDT with an operation that never applies (`deadOp_not_valid`). -/ namespace UndoableCRDTs open OpSystem UndoableCRDT /-- The nonnegative counter: states `ℕ`; `inc` always applies, `dec` only in nonzero states. -/ abbrev nnCounter : OpSystem.{0, 0} where S := ℕ P := CounterOp s0 := 0 app | .inc, n => some (n + 1) | .dec, 0 => none | .dec, n + 1 => some n /-- Axiom 1 restricted to distinct operations. -/ def OpSystem.CommutativeDistinct (C : OpSystem.{u, v}) : Prop := ∀ s p q, p ≠ q → C.Ok s [p] → C.Ok s [q] → C.Ok s [p, q] ∧ C.Ok s [q, p] ∧ C.act s [p, q] = C.act s [q, p] lemma nnCounter_finite : Finite nnCounter.P := inferInstanceAs (Finite CounterOp) lemma nnCounter_reachable : nnCounter.Reachable := by intro n refine ⟨List.replicate n .inc, ?_⟩ induction n with | zero => rfl | succ n ih => rw [List.replicate_succ', act_append_of _ ih] rfl lemma nnCounter_commutativeDistinct : nnCounter.CommutativeDistinct := by rintro s (_ | _) (_ | _) hpq hp hq · exact absurd rfl hpq · cases s with | zero => obtain ⟨t, ht⟩ := hq; cases ht | succ n => exact ⟨⟨n + 1, rfl⟩, ⟨n + 1, rfl⟩, rfl⟩ · cases s with | zero => obtain ⟨t, ht⟩ := hp; cases ht | succ n => exact ⟨⟨n + 1, rfl⟩, ⟨n + 1, rfl⟩, rfl⟩ · exact absurd rfl hpq lemma nnCounter_undoable : nnCounter.Undoable := by rintro s (_ | _) hp · exact ⟨[.dec], rfl⟩ · cases s with | zero => obtain ⟨t, ht⟩ := hp; cases ht | succ n => exact ⟨[.inc], rfl⟩ /-- The nonnegative counter violates Axiom 1 with `p = q = dec` in state `1`. -/ lemma nnCounter_not_commutative : ¬ nnCounter.Commutative := by intro h obtain ⟨⟨t, ht⟩, -⟩ := h 1 .dec .dec ⟨0, rfl⟩ ⟨0, rfl⟩ cases ht /-- Net displacement of a counter action. -/ def disp (w : List CounterOp) : ℤ := (w.map (CounterOp.val (A := ℤ))).sum lemma nnCounter_act (w : List CounterOp) (n k : ℕ) (h : nnCounter.act n w = some k) : (k : ℤ) = n + disp w := by induction w generalizing n with | nil => cases h;
