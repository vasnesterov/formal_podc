Lemma 10.4 only needs that the problem is not of type II, III or IV (type I is not used). "Equivalent" (Observation 3.2) is realised by the colour swap or the complement. -/ namespace Podc.BinaryLabeling namespace Problem /-- Type II (unsolvable): an empty constraint. -/ def IsTypeII (P : Problem) : Prop := P.W = ∅ ∨ P.B = ∅ /-- Type III (trivial): one side is unconstrained and the other non-empty. -/ def IsTypeIII (P : Problem) : Prop := (P.W.Nonempty ∧ ∀ i ≤ P.δ, i ∈ P.B) ∨ ((∀ i ≤ P.d, i ∈ P.W) ∧ P.B.Nonempty) /-- Type IV (trivial): all-0 or all-1 labels are feasible on both sides. -/ def IsTypeIV (P : Problem) : Prop := (0 ∈ P.W ∧ 0 ∈ P.B) ∨ (P.d ∈ P.W ∧ P.δ ∈ P.B) /-- Forbidden degree problems `(1⁺01⁺, 01⁺0)` (Definition 10.2). -/ def IsFD (P : Problem) : Prop := ∃ i, 0 < i ∧ i < P.d ∧ (∀ x, x ∈ P.W ↔ x ≤ P.d ∧ x ≠ i) ∧ (∀ x, x ∈ P.B ↔ 1 ≤ x ∧ x + 1 ≤ P.δ) /-- Bipartite sinkless orientation problems `(1⁺0, 01⁺)` (Definition 10.3). -/ def IsBSO (P : Problem) : Prop := (∀ x, x ∈ P.W ↔ x + 1 ≤ P.d) ∧ (∀ x, x ∈ P.B ↔ 1 ≤ x ∧ x ≤ P.δ) end Problem open Problem
