import Papers.PODC2020.RoundBasedKSetTopology.UpperBounds /-! # The second graph of Figure 1 Processes `p₁, …, p₄` are `0, …, 3`; edges `0 → 1, 0 → 2, 0 → 3` (a star) and the cycle `1 → 2 → 3 → 1`. The paper (after Corollary 3.8) claims that for the symmetric model it generates, `cov₂(S) = 3` and `γ^eq(S) = 4`, so the covering bound gives `3`-set agreement while the equal-domination bound only gives `4`-set agreement. We check these claims. -/ namespace Podc.RoundBasedKSetTopology open Cslib.HeardOf Finset /-- The edges of the second graph of Figure 1. -/ def fig1bEdges : Finset (Fin 4 × Fin 4) := {(0, 1), (0, 2), (0, 3), (1, 2), (2, 3), (3, 1)} /-- The second graph of Figure 1. -/ def fig1b : Graph (Fin 4) := fun j i => (j, i) ∈ fig1bEdges theorem card_out_fig1b (P : Finset (Fin 4)) : (out fig1b P).card = (univ.filter fun q => ∃ p ∈ P, p = q ∨ (p, q) ∈ fig1bEdges).card := by congr 1; ext q; simp [mem_out, Hears, fig1b] theorem fig1b_eqDomNum : eqDomNum fig1b = 4 := eqDomNum_eq_card_of_deaf (d := 0) (by unfold Deaf fig1b; decide) theorem fig1b_cov_two : cov 2 fig1b = 3 := by apply le_antisymm · refine (cov_le_card_out fig1b (P := {1, 2}) (by decide)).trans ?_ rw [card_out_fig1b]; decide · obtain ⟨P, hP, hcov⟩ := exists_card_out_eq_cov fig1b (i := 2) (by simp) rw [← hcov, card_out_fig1b] clear hcov revert P decide /-- The claims of Section 3.2 on the symmetric model generated by the second graph of Figure 1: `cov₂ = 3`, `γ^eq = 4`, and the covering bound `2 + (4 - 3) = 3` is solvable in one round. -/ theorem figure1_example : covSet 2 (Sym {fig1b}) = 3 ∧ eqDomNumSet (Sym {fig1b}) = 4 ∧ OneRoundSolvable ℕ 3 (closedAbove (Sym {fig1b})) := by have h1 : covSet 2 (Sym {fig1b}) = 3 := by rw [covSet_sym]; simp [covSet, fig1b_cov_two] have h2 : eqDomNumSet (Sym {fig1b}) = 4 := by rw [eqDomNumSet_sym]; simp [eqDomNumSet, fig1b_eqDomNum] refine ⟨h1, h2, ?_⟩ have := corollary_3_8 (V := ℕ) {fig1b} (i := 2) (by rw [Set.mem_Ico]; simp [eqDomNumSet, fig1b_eqDomNum]) rwa [← covSet_sym, h1] at this end Podc.RoundBasedKSetTopology
