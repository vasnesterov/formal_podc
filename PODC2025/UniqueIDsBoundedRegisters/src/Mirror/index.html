subst hℓ obtain ⟨h0, h1, h2, h3, h4⟩ := step_inv hs have : I.step D (.inv q' o) = some { D with pend := Function.update D.pend q' (some o), loc := Function.update D.loc q' (I.invoke q' (D.loc q') o) } := by simp [step, ← h.pend, h0, hall o h1] refine ⟨_, this, ?_, ?_, ?_⟩ · rw [h4, h.mem] · rw [h3]; simp [h.loc, hA.invoke p q'] · rw [h2]; simp | tau p => simp only [Label.proc] at hℓ; subst hℓ obtain ⟨h0, h1, i, o, k, hact, hm, hl⟩ := step_tau hs have hact' : I.act q' (D.loc q') = .access i o k := by rw [← h.loc, hA.act q' p, hact] have : I.step D (.tau q') = some { D with mem := Function.update D.mem i ((B i).δ (D.mem i) o).1, loc := Function.update D.loc q' (k ((B i).δ (D.mem i) o).2) } := by simp [step, ← h.pend, h0, hact'] refine ⟨_, this, ?_, ?_, ?_⟩ · rw [hm, h.mem] · rw [hl]; simp [h.mem] · rw [h1]; exact h.pend | ret p r => simp only [Label.proc] at hℓ; subst hℓ obtain ⟨h0, h1, h2, l, hact, hl⟩ := step_ret hs have hact' : I.act q' (D.loc q') = .ret r l := by rw [← h.loc, hA.act q' p, hact] have : I.step D (.ret q' r) = some { D with pend := Function.update D.pend q' none, loc := Function.update D.loc q' l } := by simp [step, ← h.pend, h0, hact'] refine ⟨_, this, ?_, ?_, ?_⟩ · rw [h2, h.mem] · rw [hl];
