### Random Flooding (Appendix C.2) -/ /-- One round of Random Flooding with bits `β`: a node with bit `true` forwards to all its neighbours, a node with bit `false` performs amnesiac flooding. -/ def rfStep {V : Type*} (G : SimpleGraph V) (β : V → Bool) (S : Config V) : Config V := {e | G.Adj e.1 e.2 ∧ Active S e.1 ∧ (β e.1 = true ∨ (e.2, e.1) ∉ S)} /-- `rfStep` on finite sets. -/ def rfStepF {V : Type*} [Fintype V] [DecidableEq V] (G : SimpleGraph V) [DecidableRel G.Adj] (β : V → Bool) (S : Finset (V × V)) : Finset (V × V) := univ.filter fun e => G.Adj e.1 e.2 ∧ (∃ w, (w, e.1) ∈ S) ∧ (β e.1 = true ∨ (e.2, e.1) ∉ S) lemma coe_rfStepF {V : Type*} [Fintype V] [DecidableEq V] (G : SimpleGraph V) [DecidableRel G.Adj] (β : V → Bool) (S : Finset (V × V)) : ((rfStepF G β S : Finset (V × V)) : Config V) = rfStep G β S := by ext ⟨u, v⟩; simp [rfStepF, rfStep, Active] /-- The 5-cycle. -/ def c5 : SimpleGraph (Fin 5) where Adj x y := x = y + 1 ∨ y = x + 1 symm := ⟨fun x y h => h.symm⟩ loopless := ⟨fun x h => by fin_cases x <;> simp_all⟩ instance : DecidableRel c5.Adj := fun x y => inferInstanceAs (Decidable (x = y + 1 ∨ y = x + 1)) /-- **The window argument of Proposition C.2 is false.** On the 5-cycle (diameter `D = 2`), start Random Flooding at node `0`, and choose the bits: all `0` except node `2` in the third round. After six rounds the only message is `(0, 1)`. Then `D + 1 = 3` rounds with all bits `1` followed by rounds with all bits `0` do not stop the flood within the `D + 2` rounds claimed by the proof (the configuration is still non-empty two rounds after the all-zeros round starts). -/ theorem random_flooding_window_false : let β₀ : Fin 5 → Bool := fun _ => false let β₂ : Fin 5 → Bool := fun x => decide (x = 2) let β₁ : Fin 5 → Bool := fun _ => true let S₆ := rfStep c5 β₀ (rfStep c5 β₀ (rfStep c5 β₀ (rfStep c5 β₂ (rfStep c5 β₀ (rfStep c5 β₀ (step c5 {0} ∅)))))) S₆ = {(0, 1)} ∧ rfStep c5 β₀ (rfStep c5 β₀ (rfStep c5 β₁ (rfStep c5 β₁ (rfStep c5 β₁ S₆)))) ≠ ∅ := by intro β₀ β₂ β₁ S₆ have h0 : step c5 {0} ∅ = ↑(stepF c5 {0} ∅) := by rw [coe_stepF]; simp simp only [S₆, h0, ← coe_rfStepF] constructor · have : rfStepF c5 β₀ (rfStepF c5 β₀ (rfStepF c5 β₀ (rfStepF c5 β₂ (rfStepF c5 β₀ (rfStepF c5 β₀ (stepF c5 {0} ∅)))))) = {(0, 1)} := by decide rw [this]; simp · have : rfStepF c5 β₀ (rfStepF c5 β₀ (rfStepF c5 β₁ (rfStepF c5 β₁ (rfStepF c5 β₁ (rfStepF c5 β₀ (rfStepF c5 β₀ (rfStepF c5 β₀ (rfStepF c5 β₂ (rfStepF c5 β₀ (rfStepF c5 β₀ (stepF c5 {0} ∅))))))))))) ≠ ∅ := by decide exact_mod_cast this /-!
