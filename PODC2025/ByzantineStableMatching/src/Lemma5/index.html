import Papers.PODC2025.ByzantineStableMatching.Groups /-! # Lemma 5: the fully-connected unauthenticated lower bound Paper: for `n = 6` (`k = 3`) no protocol achieves sSM with `tL = tR = 1` in a fully-connected unauthenticated network; proof by a 12-node double cover (Fig. 2). We prove directly, for every `k ≥ 2` and every network `G` (in particular the fully-connected one), that sSM is unsolvable when `3 tL ≥ k` and `3 tR ≥ k`. Each side is split into three groups `a, b, c` (resp. `u, v, w`) of size `≤ ⌈k/3⌉` (the groups `b`, `u` may be empty, which covers `k = 2` where the paper's route via Lemma 3 does not apply). The double cover is the paper's: each party has two copies; the edges `a–c`, `a–w`, `c–u`, `u–w` (between groups with labels `0` and `2`) cross between the copies, all other edges stay inside a copy. (We read off this voltage assignment from the three honest sets of Fig. 2 (ii)–(iv), which pins it down uniquely.) -/ namespace Papers.PODC2025.ByzantineStableMatching open Cslib.Byzantine.LockStep Sum variable {k : ℕ} namespace Lemma5 /-- Edges between groups labelled `0` and `2` cross between the two copies. -/ def volt (a b : Fin 3) : Bool := (a = 0 && b = 2) || (a = 2 && b = 0) /-- The double cover of Lemma 5. Node `(p, c)` is copy `c` of party `p`. All `L`-nodes have the `v`-representative `jV` as favourite; `R`-nodes of copy `true` favour `iA`, of copy `false` favour `iC`. -/ def cover (gL gR : Fin k → Fin 3) (iA iC jV : Fin k) : CoverSystem (Party k) (Fin k) (Party k × Bool) := CoverSystem.doubleCover (fun q c p => some (xor c (volt (label gL gR q) (label gL gR p)))) fun q => match q with | (inl _, _) => jV | (inr _, c) => if c then iA else iC /-- The general form of Lemma 5. -/ theorem not_solvable_of_groups (G : SimpleGraph (Party k)) {tL tR : ℕ} (gL gR : Fin k → Fin 3) (iA iC jV : Fin k) (hA : gL iA = 0) (hC : gL iC = 2) (hV : gR jV = 1) (hgL : ∀ z, {i | gL i = z}.ncard ≤ tL) (hgR : ∀ z, {j | gR j = z}.ncard ≤ tR) : ¬ Solvable G (thresholds k tL tR) (sSM k) := by rintro ⟨M, alg, halg⟩ set C := cover gL gR iA iC jV set lab := label gL gR have hZ : ∀ z : Fin 3, {p | lab p = z} ∈ thresholds k tL tR := fun z => label_mem_thresholds gL gR {z} (hgL z) (hgR z) -- Scenario (ii): groups `c`, `w` Byzantine, copy `true` honest. have h2 : C.Decides G alg (inl iA, true) (some jV) := by have hL : C.Lift G {p | lab p = 2} (fun p => (p, true)) := CoverSystem.doubleCover_lift fun p q hp hq _ => by have key : ∀ a b : Fin 3, a ≠ 2 → b ≠ 2 → xor true (volt a b) = true := by decide exact congrArg some (key _ _ hq hp) have hs := halg (C.scenario G alg {p | lab p = 2} (fun p => (p, true)) fun _ => jV) (hZ 2) (fun _ _ _ h => h) have hiA : inl iA ∉ {p | lab p = 2} := by simp [lab, hA] have hjV : inr jV ∉ {p | lab p = 2} := by simp [lab, hV] have := (hs.2 iA jV hiA hjV (by rw [C.scenario_input_of_not_mem _ hiA];
