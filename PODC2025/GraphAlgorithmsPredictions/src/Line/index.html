import Papers.PODC2025.GraphAlgorithmsPredictions.Basic import Cslib.Computability.Distributed.Local.Locality -- CSLib candidate: Cslib/Computability/Distributed/Local/Line.lean (oriented line in the LOCAL model) /-! # The oriented line in the LOCAL model Model of Section 6 of the full version (arXiv:2501.05267) of Boyar–Ellen–Larsen: `n` nodes connected in a line; every node knows its own identifier and the identifiers of its left and right neighbours (if they exist). Identifiers are arbitrary natural numbers (no bound `d` is fixed). Following CSLib's LOCAL locality (`Cslib.Local.IsLocal`), nodes are named by their identifiers: an instance is an injective placement `id : Fin n ↪ ℕ` of identifiers on positions `0, …, n - 1` (left to right); the network is `lineG id : SimpleGraph ℕ` (the path, mapped along `id`) and the input label of node `u` is `lineLab id u = (left neighbour, right neighbour)`. A `T`-round algorithm is any `f : SimpleGraph ℕ → (ℕ → Lab) → ℕ → β` with `IsLocal T f`. * `ix id k`: identifier at position `k`. * `lineG_adj_ix`, `lineLab_ix`: adjacency and labels in terms of positions. * `exists_pos_of_edist_le`, `exists_pos_of_edist_lt`: a node at distance `≤ m` from position `c` sits at a position within `m` of `c`. * `viewEq_window`: two lines that carry the same identifiers on a window of `2T + 3` consecutive positions give the same radius-`T` view to the centre of the window. -/ namespace Podc.GraphAlgorithmsPredictions open SimpleGraph Cslib.Local /-- The initial knowledge of a node of the line besides its own identifier: the identifiers of its left and right neighbours (`none` at the ends). -/ abbrev Lab := Option ℕ × Option ℕ section Line variable {n : ℕ} (id : Fin n ↪ ℕ) /-- The identifier at position `k` of the line (junk value `0` for `k ≥ n`). -/ def ix (k : ℕ) : ℕ := if h : k < n then id ⟨k, h⟩ else 0 lemma ix_of_lt {k : ℕ} (h : k < n) : ix id k = id ⟨k, h⟩ := by simp [ix, h] @[simp] lemma ix_fin (i : Fin n) : ix id i = id i := ix_of_lt id i.2 lemma ix_inj {k l : ℕ} (hk : k < n) (hl : l < n) (h : ix id k = ix id l) : k = l := by rw [ix_of_lt id hk, ix_of_lt id hl] at h exact congrArg Fin.val (id.injective h) /-- The oriented line with identifiers `id` (position `i` is adjacent to `i ± 1`), as a graph on the identifiers. -/ def lineG : SimpleGraph ℕ := (pathGraph n).map id lemma lineG_adj_ix {q : ℕ} (hq : q < n) (y : ℕ) : (lineG id).Adj (ix id q) y ↔ (q + 1 < n ∧ y = ix id (q + 1)) ∨ (0 < q ∧ y = ix id (q - 1)) := by unfold lineG rw [map_adj] constructor · rintro ⟨u, v, huv, hu, rfl⟩ rw [pathGraph_adj] at huv have hq' : u.val = q := ix_inj id u.2 hq (by rw [ix_fin, hu]) rcases huv with h | h · exact Or.inl ⟨by omega, by rw [← ix_fin];
