rw [ha, hb] at hth; rw [ha, hb] at hne exact absurd (congrArg t (ht hth)) hne namespace STM variable (X : STM Tx O P) /-- The execution of `σ` is well-formed and has no abort. -/ def Good (σ : List (Inv Tx O)) : Prop := WF (X.hist σ) ∧ AllSucc (X.hist σ) lemma good_nil : X.Good [] := ⟨WF.nil, fun _ h => by simp at h⟩ lemma okNext_of_S {σ : List (Inv Tx O)} {i : Inv Tx O} (hs : AllSucc (X.hist σ)) (hok : OKNextS σ i) : OKNext (X.hist σ) i := by obtain ⟨h1, h2, h3⟩ := hok refine ⟨?_, h2, ?_⟩ · rw [terminated_iff_of_succ hs, map_inv_hist]; exact hs.of_append refine RT.of_split (X.hist σ₁) (h₂ ++ [⟨i, r⟩]) (by rw [hh]; simp) ?_ ?_ ?_ · rw [committed_iff_of_succ hs₁, map_inv_hist]; exact hA · rw [began_iff_map, map_inv_hist]; rintro ⟨j, hj, hjt⟩; exact hB j hj hjt · exact ⟨⟨i, r⟩, by simp, rfl⟩ /-- One more operation of a good execution succeeds, if it is well-formed and the transaction is read-only or every update transaction that has begun committed before it began. -/ lemma good_snoc (hm : X.MVPermissive) {σ : List (Inv Tx O)} {i : Inv Tx O} (hg : X.Good σ) (hok : OKNextS σ i) (hni : ro i.tx = true ∨ ∀ T', T' ≠ i.tx → ro T' = false → (∃ j ∈ σ, Inv.tx j = T') → ∃ σ₁ σ₂, σ = σ₁ ++ σ₂ ∧ Inv.commit T' ∈ σ₁ ∧ ∀ j ∈ σ₁, Inv.tx j ≠ i.tx) : X.Good (σ ++ [i]) := by have hok' := X.okNext_of_S hg.2 hok have hne : X.resp σ i ≠ none := by refine X.resp_ne_none hm σ i hg.1 hok' ?_ rcases hni with h | h · exact Or.inl h · refine Or.inr fun T' hT' hro ⟨hc, _⟩ => ?_ have hb : ∃ j ∈ σ, Inv.tx j = T' := by have := hc.2.1 rw [began_append, began_iff_map, map_inv_hist] at this rcases this with h' | ⟨e, he, het⟩ · exact h' · simp at he;
