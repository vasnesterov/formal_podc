import Papers.PODC2010.MultiVersionSTM.Snapshot /-! # Lemmas about the snapshot STMs Computation rules and index-level characterizations used in `SnapshotProofs.lean`: * list helpers relating decompositions `h = h₁ ++ e :: h₂` and indices (`h[k]?`, `h.take k`); * `latest`: `latest_nil`, `latest_snoc`, `latest_cases` (the last `CommitWrites` event), `latest_eq_latest_take` (no overwrite in between ⇒ same latest version); * `frz`: `frz_eq_some_iff`, `frz_eq_none_iff`, stability under extension; * `firstIdx` and `snap` basics; * `Gen lazy h`: `h` is generated by the strategy `snapResp lazy`; * index-level facts about well-formed histories, and `cidx h T`, the index of the commit of `T`. -/ namespace Podc.MultiVersionSTM open scoped Classical variable {Tx O P : Type} /-! ### List helpers -/ section ListHelpers variable {α : Type} {l l₁ l₂ : List α} {a x : α} {k : ℕ} lemma eq_take_append_cons (hk : l[k]? = some a) : l = l.take k ++ a :: l.drop (k + 1) := by obtain ⟨hlt, rfl⟩ := List.getElem?_eq_some_iff.1 hk conv_lhs => rw [← List.take_append_drop k l] rw [List.drop_eq_getElem_cons hlt] lemma getElem?_of_decomp (h : l = l₁ ++ a :: l₂) : l[l₁.length]? = some a := by subst h; simp lemma take_of_decomp (h : l = l₁ ++ a :: l₂) : l.take l₁.length = l₁ := by subst h; simp lemma lt_length_of_getElem? ### Histories generated by the snapshot strategy -/ section Gen variable [TxSys Tx P] /-- `h` is generated by the strategy `snapResp lazy`: every response is computed from the history before it. -/ def Gen (lazy : Bool) (h : Hist Tx O) : Prop := ∀ h₁ e h₂, h = h₁ ++ e :: h₂ → e.resp = snapResp lazy h₁ e.inv variable {lazy : Bool} {h : Hist Tx O} lemma Gen.nil : Gen lazy ([] : Hist Tx O) := by intro h₁ e h₂ he; simp at he lemma Gen.snoc (hg : Gen lazy h) (i : Inv Tx O) : Gen lazy (h ++ [⟨i, snapResp lazy h i⟩]) := by intro a e' b hab rcases List.eq_nil_or_concat b with rfl | ⟨b', x, rfl⟩ · obtain ⟨rfl, rfl⟩ : h = a ∧ (⟨i, snapResp lazy h i⟩ : Event Tx O) = e' := by simpa using hab rfl · have : h = a ++ e' :: b' := by have := hab; simp only [List.concat_eq_append] at this rw [show a ++ e' :: (b' ++ [x]) = (a ++ e' :: b') ++ [x] by simp] at this exact (List.append_inj' this rfl).1 exact hg a e' b' this lemma Gen.at (hg : Gen lazy h) {k : ℕ} {e : Event Tx O} (hk : h[k]? = some e) : e.resp = snapResp lazy (h.take k) e.inv := hg _ e _ (eq_take_append_cons hk) lemma Gen.of_append {h' : Hist Tx O} (hg : Gen lazy (h ++ h')) : Gen lazy h := by intro a e b hab exact hg a e (b ++ h') (by simp [hab]) /-- The histories of the snapshot STM are generated by its strategy. -/ lemma gen_hist (lazy : Bool) (σ : List (Inv Tx O)) : Gen lazy ((snapSTM (Tx := Tx) (O := O) (P := P) lazy).hist σ) := by induction σ using List.reverseRecOn with | nil => exact Gen.nil | append_singleton σ i ih => rw [snapSTM, ofStrategy.hist_snoc] exact Gen.snoc ih i end Gen /-! ### Well-formed histories, by index; the commit index `cidx` -/ section WFIdx variable {h : Hist Tx O} {T W : Tx} {o : O} {k : ℕ} {e : Event Tx O} lemma committed_iff_getElem? : Committed h T ↔ ∃ (k : ℕ) (r : Option Tx), h[k]? = some ⟨.commit T, some r⟩ := by simp only [Committed, List.mem_iff_getElem?] constructor · rintro ⟨r, i, hi⟩; exact ⟨i, r, hi⟩ · rintro ⟨i, r, hi⟩; exact ⟨r, i, hi⟩ /-- The index of the first successful commit of `T` in `h` (`h.length` if there is none). -/ noncomputable def cidx (h : Hist Tx O) (T : Tx) : ℕ := h.findIdx (fun e => decide (∃ r, e = ⟨.commit T, some r⟩)) lemma cidx_le_length : cidx h T ≤ h.length := List.findIdx_le_length lemma cidx_le_of {r : Option Tx} (hk : h[k]? = some ⟨.commit T, some r⟩) : cidx h T ≤ k := by by_contra!
