import Mathlib import Cslib.Computability.Distributed.SharedMemory.Model import Cslib.Computability.Distributed.SharedMemory.LinPoints /-! # Jayanti–Tarjan, *A Randomized Concurrent Algorithm for Disjoint Set Union* (PODC 2016): the object and the algorithms of Section 3 * `dsu E` — the sequential specification of disjoint set union over a finite set of elements `E`: the state is the "same set" relation (initially equality), `sameSet x y` returns whether `x` and `y` are related, `unite x y` merges their classes (and returns the dummy response `true`). * Base objects: one compare&swap object per element `a`, holding `a.parent`, initially `a` (`casObject E a`). The algorithms use only `read` and `cas`. * `FindKind` — the three versions of `Find`: Algorithm 1 (`naive`, no compaction) and Algorithm 4 / 5 (`split 0` = one-try splitting, `split 1` = two-try splitting; in general `split k` tries every parent update `k + 1` times). * `dsuImpl K key P` — Algorithms 2 (`SameSet`) and 3 (`Unite`) with `Find` of kind `K`, linking by the (fixed, injective) node ids `key : E → ℕ` (the paper's random total order). Every primitive access is one atomic step; local computation is folded into the continuation of the preceding access (as in CSLib's model). -/ namespace Podc.ConcurrentDSU open Cslib.SharedMemory variable {E : Type} /-- Operations of the disjoint set union object. -/ inductive DSUOp (E : Type) where /-- `SameSet(x, y)`: are `x` and `y` in the same set? -/ | sameSet (x y : E) /-- `Unite(x, y)`: unite the sets containing `x` and `y`. -/ | unite (x y : E) /-- The relation obtained from `s` by merging the classes of `x` and `y` (for an equivalence `s`, this is the equivalence generated by `s` and the pair `(x, y)`). -/ def joinRel (s : E → E → Prop) (x y : E) : E → E → Prop := fun a b => s a b ∨ (s a x ∧ s y b) ∨ (s a y ∧ s x b) /-- The sequential specification of disjoint set union over `E`: the state is the relation "same set", initially equality. `sameSet` is read-only and answers the membership query; `unite` merges the two classes and returns the dummy acknowledgement `true`. -/ noncomputable abbrev dsu (E : Type) : ObjSpec where State := E → E → Prop Op := DSUOp E Resp := Bool init := Eq δ s o := match o with | .sameSet x y => (s, @decide (s x y) (Classical.dec _)) | .unite x y => (joinRel s x y, true) @[simp] lemma dsu_δ_sameSet (s : E → E → Prop) (x y : E) : (dsu E).δ s (.sameSet x y) = (s, @decide (s x y) (Classical.dec _)) := rfl @[simp] lemma dsu_δ_unite (s : E → E → Prop) (x y : E) : (dsu E).δ s (.unite x y) = (joinRel s x y, true) := rfl @[simp] lemma dsu_init : (dsu E).init = Eq := rfl /-- The parent pointers: element `a` is a compare&swap object over `E`, initially `a`. -/ abbrev Ptr (E : Type) [DecidableEq E] (a : E) : ObjSpec := casObject E a /-- The version of `Find`. `split k` tries every parent update `k + 1` times. -/ inductive FindKind where /-- Algorithm 1: no compaction. -/ | naive /-- Splitting with `k + 1` tries per node (`split 0`: Algorithm 4, `split 1`: Algorithm 5). -/ | split (k : ℕ) /-- One-try splitting (Algorithm 4). -/ abbrev FindKind.oneTry : FindKind := .split 0 /-- Two-try splitting (Algorithm 5). -/ abbrev FindKind.twoTry : FindKind := .split 1 /-- Number of tries per node of a splitting `Find`. -/ def FindKind.tries : FindKind → ℕ | .naive => 1 | .split k => k + 1 lemma FindKind.tries_pos (K : FindKind) : 0 < K.tries := by cases K <;> simp [tries] /-- Local states of a `Find` (program counters with their variables). * Algorithm 1: `tst u` — evaluate `u = u.parent`; `adv u` — `u ← u.parent`. * Algorithms 4/5: `rdP u i` — `v ← u.parent`; `rdG u v i` — `w ← v.parent` (then return `v` if `v = w`); `cas u v w i` — `Cas(u.parent, v, w)`; `i` is the number of the current try. -/ inductive FS (E : Type) where | tst (u : E) | adv (u : E) | rdP (u : E) (i : ℕ) | rdG (u v : E) (i : ℕ) | cas (u v w : E) (i : ℕ) /-- The next primitive of a `Find`: a read of `a.parent` (continuation gets the value read) or a `Cas` (whose result `Find` ignores). A continuation returns either the next `Find` state (`inl`) or the root found (`inr`). -/ inductive FStep (E : Type) where | read (a : E) (k : E → FS E ⊕ E) | cas (a old new : E) (k : FS E ⊕ E) variable [DecidableEq E] /-- Initial state of `Find(x)`. -/ def FS.start : FindKind → E → FS E | .naive, x => .tst x | .split _, x => .rdP x 0 /-- The program of `Find` (Algorithms 1, 4, 5). -/ def FS.next (K : FindKind) : FS E → FStep E | .tst u => .read u (fun r => if r = u then .inr u else .inl (.adv u)) | .adv u => .read u (fun r => .inl (.tst r)) | .rdP u i => .read u (fun v => .inl (.rdG u v i)) | .rdG u v i => .read v (fun w => if w = v then .inr v else .inl (.cas u v w i)) | .cas u v w i => .cas u v w (if i + 1 < K.tries then .inl (.rdP u (i + 1)) else .inl (.rdP v 0)) /-- Local states of a process. * `fu b fs v` — running `u ← Find(u)` (state `fs`), `v` is the current value of `v`; * `fv b u fs` — running `v ← Find(v)`; * `chk u v` — `SameSet`, line 8: test `u = u.parent`; * `lnk u v` — `Unite`, lines 8–10: the `Cas` linking the smaller of `u`, `v` to the other; * `done r` — about to return `r`. `b = true` for `Unite`, `false` for `SameSet`. -/ inductive Loc (E : Type) where | idle | fu (b : Bool) (fs : FS E) (v : E) | fv (b : Bool) (u : E) (fs : FS E) | chk (u v : E) | lnk (u v : E) | done (r : Bool) /-- Decoding the response of a `read` of `a.parent` (the `inr` case does not occur). -/ def val (d : E) : E ⊕ Bool → E | .inl a => a | .inr _ => d /-- What happens when `Find(u)` of the loop body continues or returns `r`. -/ def contU (K : FindKind) (b : Bool) (v : E) : FS E ⊕ E → Loc E | .inl fs => .fu b fs v | .inr r => .fv b r (FS.start K v) /-- What happens when `Find(v)` of the loop body continues or returns `r` (line 7 of both algorithms: return `true` if `u = v`; otherwise `SameSet` goes to line 8, `Unite` to the link). -/ def contV (b : Bool) (u : E) : FS E ⊕ E → Loc E | .inl fs => .fv b u fs | .inr r => if u = r then .done true else if b then .lnk u r else .chk u r variable {P : Type} /-- Embedding one `Find` step into a process step. -/ def liftF (K : FindKind) (fs : FS E) (c : FS E ⊕ E → Loc E) : Action (dsu E) (Ptr E) (Loc E) := match fs.next K with | .read a k => .access a .read (fun r => c (k (val a r))) | .cas a o n k => .access a (.cas o n) (fun _ => c k) /-- The program of a process (Algorithms 2 and 3 with `Find` of kind `K`). -/ def act (K : FindKind) (key : E → ℕ) : Loc E → Action (dsu E) (Ptr E) (Loc E) | .idle => .ret true .idle | .fu b fs v => liftF K fs (contU K b v) | .fv b u fs => liftF K fs (contV b u) | .chk u v => .access u .read (fun r => if val u r = u then .done false else .fu false (FS.start K u) v) | .lnk u v => if key u < key v then .access u (.cas u v) (fun r => if r = .inr true then .done true else .fu true (FS.start K u) v) else .access v (.cas v u) (fun r => if r = .inr true then .done true else .fu true (FS.start K u) v) | .done r => .ret r .idle /-- Invocation: `SameSet(x, y)` / `Unite(x, y)` start with `u ← x`, `v ← y`, `u ← Find(u)`. -/ def invoke (K : FindKind) : Loc E → DSUOp E → Loc E | _, .sameSet x y => .fu false (FS.start K x) y | _, .unite x y => .fu true (FS.start K x) y /-- The concurrent disjoint set union implementation of Section 3 (Algorithms 2 and 3, `Find` of kind `K`), for processes `P`, linking by the node ids `key`. -/ def dsuImpl (K : FindKind) (key : E → ℕ) (P : Type) : Impl (dsu E) (Ptr E) P (Loc E) where memInit a := a locInit _ := .idle invoke _ l o := invoke K l o act _ l := act K key l end Podc.ConcurrentDSU
