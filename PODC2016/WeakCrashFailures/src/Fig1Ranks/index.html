import Papers.PODC2016.WeakCrashFailures.Fig1Safety /-! # Fig. 1: levels and ranks for the termination proof * `lvl c`: the position of a program counter in the code (rounds, line 16 check, line 21 loop); it never decreases (`lvl_step`). * `rk c`: progress inside the non-waiting parts of a level; every own step that stays on the same level decreases it (`rk_step`), provided `decision = ⊥` and, in the last loop, all states are `3`. * `wrk S0 c`: progress inside the waiting loop of round `r`, relative to a set `S0` of processes with `state ≥ r` of size `≥ n − 1` (`wrk_step`). * `drk c`: progress towards the next read of `decision` (`drk_step`), used once `decision ≠ ⊥`. -/ namespace Podc.WeakCrashFailures.Fig1 open Finset variable {n : ℕ} {x : Fin n → ℕ} /-- The level of a program counter. -/ def lvl : PC → ℕ | .start => 0 | .scan r _ _ _ => 2 * r - 1 | .scanIn r _ _ _ => 2 * r - 1 | .wait r _ _ _ => 2 * r | .chk _ _ _ => 7 | .chkIn _ _ _ => 7 | .fin _ _ => 8 | .finDec _ _ _ => 8 | .finIn _ _ => 8 | .done _ => 9 /-- Rank inside the non-waiting code of a level. -/ def rk (n : ℕ) : PC → ℕ | .scan _ j _ _ => 2 * (n - j) + 2 | .scanIn _ j _ _ => 2 * (n - j) + 1 | .chk j _ _ => 2 * (n - j) + 2 | .chkIn j _ _ => 2 * (n - j) + 1 | .fin j _ => 3 * (n - j) + 2 | .finDec j _ b => 3 * (n - j) + if b then 1 else 3 | .finIn j _ => 3 * (n - j) | _ => 0 /-- Rank inside a waiting loop relative to `S0`. -/ def wrk (n : ℕ) (S0 : Finset (Fin n)) : PC → ℕ | .wait _ j cnt _ => if (S0.filter fun k => k.val < j).card ≤ cnt then n - j else 2 * n + 1 - j | _ => 0 /-- Rank towards the next read of `decision`. -/ def drk (n : ℕ) : PC → ℕ | .start => 3 * n + 4 | .scan _ j _ _ => 2 * (n - j) + n + 3 | .scanIn _ j _ _ => 2 * (n - j) + n + 2 | .wait _ j _ _ => n - j + 1 | .chk j _ _ => 2 * (n - j) + 4 | .chkIn j _ _ => 2 * (n - j) + 3 | .fin _ _ => 2 | .finDec _ _ _ => 1 | .finIn _ _ => 3 | .done _ => 0 lemma lvl_step {C : Cfg n} (hI : Inv x C) (q : Fin n) : lvl (C.pc q) ≤ lvl ((step x q C).pc q) := by have hli := hI.li q cases hc : C.pc q with | start => simp [step, hc, lvl] | scan r j cnt mx => rw [hc] at hli; obtain ⟨h1, h3, hjn, -⟩ := hli simp only [step, hc] at hlv ⊢ by_cases hj : j < n · simp only [hj, ite_true, setPc_pc_self, wrk] rw [card_filter_lt_succ S0 hj] by_cases hm : (⟨j, hj⟩ : Fin n) ∈ S0 · have hr : r ≤ stAt C j := by rw [stAt_of_lt hj];
