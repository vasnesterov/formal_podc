import Papers.PODC2016.WeakCrashFailures.Fig1Model /-! # Fig. 1: the inductive invariant The invariant `Inv x C` consists of a local invariant `LI` per process (by program counter) and global facts. The core of agreement (Lemmas 3 and 4 of the paper) is the notion of a *tag* (`Tag`): every value ever decided or written into `decision` is `val x t` for a tag `t` with `TagFact C t`, where * `Tag.ex y` stands for "the maximum of the inputs of all processes but `y`" (lines 7 and 20): `y`'s state is smaller than everybody else's, and `y` is frozen: it has terminated, or it is *doomed* (`DoomedR C 3 y`: all other processes have `state = 3`, and `y` has not yet scanned in round 3 in a way that would let it write `3`); * `Tag.all` stands for "the maximum of all inputs" (line 25): every state is `3`. Two tags satisfying `TagFact` in the same configuration are equal (`TagFact.unique`, `n ≥ 2`), which gives agreement. -/ namespace Podc.WeakCrashFailures.Fig1 open Finset variable {n : ℕ} /-- Number of processes with index `< j` other than `y`. -/ def cntLt (j : ℕ) (y : Fin n) : ℕ := (univ.filter fun k : Fin n => k.val < j ∧ k ≠ y).card /-- Number of processes with `state ≥ r`. -/ def cntGE (C : Cfg n) (r : ℕ) : ℕ := (univ.filter fun k : Fin n => r ≤ C.st k).card /-- The indices `< j`. -/ def ltSet (j : ℕ) : Finset (Fin n) := univ.filter fun k : Fin n => k.val < j /-- `y` has not yet passed (or is about to fail) the line-7 test of round `r`: it is in an earlier round, or in the scan of round `r` having counted exactly the other processes read so far. -/ def PreRpc (r : ℕ) (y : Fin n) : PC → Prop | .start => True | .scan r' j cnt _ => r' < r ∨ (r' = r ∧ cnt = cntLt j y) | .scanIn r' j cnt _ => r' < r ∨ (r' = r ∧ cnt = cntLt (j + 1) y) | .wait r' _ _ _ => r' < r | .done _ => True | _ => False /-- `y` is doomed in round `r`: all other processes have `state ≥ r`, `y`'s state is `< r`, and `y` is before (or within, counting everybody else) the scan of round `r`. Such a process never writes `r` into its state register: it decides in line 7 of round `r` at the latest. -/ def DoomedR (C : Cfg n) (r : ℕ) (y : Fin n) : Prop := (∀ k, k ≠ y → r ≤ C.st k) ∧ C.st y < r ∧ PreRpc r y (C.pc y) /-- Tags of decision values. -/ inductive Tag (n : ℕ) | ex (y : Fin n) | all /-- The value of a tag. -/ def val (x : Fin n → ℕ) : Tag n → ℕ | .ex y => (univ.erase y).sup x | .all => univ.sup x /-- The facts justifying a tag. -/ def TagFact (C : Cfg n) : Tag n → Prop | .ex y => (∀ k, k ≠ y → C.st y < C.st k) ∧ ((∃ v, C.pc y = .done v) ∨ DoomedR C 3 y) | .all => ∀ k, C.st k = 3 /-- The local invariant of process `i` with program counter `c`. -/ def LIpc (x : Fin n → ℕ) (C : Cfg n) (i : Fin n) : PC → Prop | .start => C.st i = 0 | .scan r j cnt mx => 1 ≤ r ∧ r ≤ 3 ∧ j ≤ n ∧ C.st i + 1 = r ∧ (r = 2 → n - 1 ≤ cntGE C 1) ∧ ∃ S : Finset (Fin n), (∀ k ∈ S, k.val < j ∧ r ≤ C.st k) ∧ S.card = cnt ∧ mx = S.sup x | .scanIn r j cnt mx => 1 ≤ r ∧ r ≤ 3 ∧ ∃ h : j < n, C.st i + 1 = r ∧ (r = 2 → n - 1 ≤ cntGE C 1) ∧ r ≤ C.st ⟨j, h⟩ ∧ ∃ S : Finset (Fin n), (∀ k ∈ S, k.val < j ∧ r ≤ C.st k) ∧ S.card + 1 = cnt ∧ mx = S.sup x | .wait r j cnt mx => 1 ≤ r ∧ r ≤ 3 ∧ j ≤ n ∧ C.st i = r ∧ (∃ S : Finset (Fin n), (∀ k ∈ S, k.val < j ∧ r ≤ C.st k) ∧ S.card = cnt) ∧ ∃ T : Finset (Fin n), mx = T.sup x | .chk j cnt mx => j ≤ n ∧ C.st i = 3 ∧ (∃ S3 : Finset (Fin n), S3.card = n - 1 ∧ ∀ k ∈ S3, C.st k = 3) ∧ ∃ Cn : Finset (Fin n), (∀ k ∈ Cn, k.val < j ∧ 2 ≤ C.st k) ∧ Cn.card = cnt ∧ mx = Cn.sup x ∧ ∀ k : Fin n, k.val < j → k ∉ Cn → DoomedR C 3 k | .chkIn j cnt mx => C.st i = 3 ∧ ∃ h : j < n, 2 ≤ C.st ⟨j, h⟩ ∧ (∃ S3 : Finset (Fin n), S3.card = n -
