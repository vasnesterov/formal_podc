import Papers.PODC2016.CollaborativeListSpec.Model /-! # Generic happens-before facts Facts about happens-before in executions (`Model.lean`) used for the correctness proof of RGA, independent of the protocol: * `HB.exists_send`: happens-before leaving a replica goes through a later send of that replica. * `Known α m r q`: event `q` happens before (or is) some event of replica `r` among the first `m` events. How it evolves with one more event (`known_succ_*`); for a receive this uses causal broadcast (`known_succ_recv`). * `Window α m r q`: `q` is an event of `r` after the last send of `r` before position `m` (the events whose effects are in the send buffer). -- CSLib candidate: generic facts on happens-before / causal broadcast. -/ namespace Podc.CollaborativeListSpec variable {Rep Op : Type} /-- Every receive of `α` refers to a send event (a consequence of well-formedness). -/ def RecvOfSend (α : List (Act Rep Op)) : Prop := ∀ c r s, α[c]? = some (Act.recv r s) → s < c ∧ ∃ r', α[s]? = some (Act.send r') /-- Event `q` is `≤hb` some event of replica `r` among the first `m` events. -/ def Known (α : List (Act Rep Op)) (m : ℕ) (r : Rep) (q : ℕ) : Prop := ∃ e a, e < m ∧ α[e]? = some a ∧ a.rep = r ∧ (q = e ∨ HB α q e) /-- Event `q` is an event of `r` before position `m` with no send of `r` in between. -/ def Window (α : List (Act Rep Op)) (m : ℕ) (r : Rep) (q : ℕ) : Prop := q < m ∧ (∃ a, α[q]? = some a ∧ a.rep = r) ∧ ∀ j, q < j → j < m → α[j]? ≠ some (Act.send r) variable {α : List (Act Rep Op)} /-- Happens-before from `i` to an event of another replica goes through a send of the replica of `i` at or after `i`. -/ theorem HB.exists_send (hR : RecvOfSend α) {i j : ℕ} (h : HB α i j) {a b : Act Rep Op} (ha : α[i]? = some a) (hb : α[j]? = some b) (hab : a.rep ≠ b.rep) : ∃ s, i ≤ s ∧ α[s]? = some (Act.send a.rep) ∧ HB α s j := by induction h using Relation.TransGen.head_induction_on generalizing a with | single h => rename_i i obtain ⟨hij, a', b', ha', hb', hrel⟩ := h rw [ha, Option.some.injEq] at ha'; subst ha' rw [hb, Option.some.injEq] at hb'; subst hb' rcases hrel with hrel | hrel · exact absurd hrel hab · cases b with | doOp => cases hrel | send => cases hrel | recv r s => simp only [Act.src?_recv, Option.some.injEq] at hrel; subst hrel obtain ⟨-, r', hr'⟩ := hR _ _ _ hb rw [ha, Option.some.injEq] at hr'; subst hr' exact ⟨s, le_rfl, ha, Relation.TransGen.single ⟨hij, _, _, ha, hb, Or.inr rfl⟩⟩ | head h' h ih => rename_i i c obtain ⟨hic, a', c', ha', hc', hrel⟩ := h' rw [ha, Option.some.injEq] at ha'; subst ha' by_cases hac : c'.rep = a.rep · obtain ⟨s, hcs, hs, hsj⟩ := ih hc' (by rwa [hac]) exact ⟨s, by omega, by rw [hs, hac], hsj⟩ · rcases hrel with hrel | hrel · exact absurd hrel.symm hac · cases c' with | doOp => cases hrel | send => cases hrel | recv r s => simp only [Act.src?_recv, Option.some.injEq] at hrel; subst hrel obtain ⟨-, r', hr'⟩ := hR _ _ _ hc' rw [ha, Option.some.injEq] at hr'; subst hr' exact ⟨s, le_rfl, ha, Relation.TransGen.head ⟨hic, _, _, ha, hc', Or.inr rfl⟩ h⟩ theorem Known.mono {m m' : ℕ} {r : Rep} {q : ℕ} (h : Known α m r q) (hm : m ≤ m') : Known α m' r q := by obtain ⟨e, a, he, hea, har, hq⟩ := h exact ⟨e, a, by omega, hea, har, hq⟩ theorem Known.of_hb {m : ℕ} {r : Rep} {q q' : ℕ} (h : Known α m r q) (hq : HB α q' q) : Known α m r q' := by obtain ⟨e, a, he, hea, har, hqe⟩ := h refine ⟨e, a, he, hea, har, Or.inr ?_⟩ rcases hqe with rfl | hqe · exact hq · exact hq.trans hqe theorem known_succ_of_ne {m : ℕ} {r : Rep} {a : Act Rep Op} (hm : α[m]? ≠ some (Act.send rq) · exact Or.inr ⟨hqk.lt, ⟨_, hq, rfl⟩, hw⟩ · push Not at hw obtain ⟨j, hqj, hjk, hj⟩ := hw exact Or.inl (recv_of j hj hqj (HB.of_rep hjk hj hsrc rfl)) · obtain ⟨s, hqs, hs, hsk⟩ := hqk.exists_send hR hq hsrc (by simpa using hrq) rcases Nat.lt_or_eq_of_le hqs with hqs | rfl · exact Or.inl (recv_of s hs hqs hsk) · rw [hs] at hq; cases hq · rintro (h | ⟨hqs, ⟨b, hb, hbr⟩, -⟩) · exact h.mono (by omega) · exact ⟨m, _, by omega, hm, rfl, Or.inr ((HB.of_rep hqs hb hsrc (by simpa using hbr)).trans (HB.of_recv (hR _ _ _ hm).1 hsrc hm))⟩ theorem window_succ_of_ne {m : ℕ} {r : Rep} {a : Act Rep Op} (hm : α[m]?
