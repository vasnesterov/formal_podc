(3 * E.d + 5)) / 2 = t by omega] theorem act_Bee (h : E.odd = true) : E.act w (5 * E.d + 7) = .doOp E.r1 (.ins E.Bee 0) := by rw [act, if_neg (by omega), if_neg (by omega), if_neg (by omega), if_neg (by omega), if_pos ⟨h, rfl⟩] theorem act_delBee (h : E.odd = true) : E.act w (5 * E.d + 8) = .doOp E.r1 (.del E.Bee) := by rw [act, if_neg (by omega), if_neg (by omega), if_neg (by omega), if_neg (by omega), if_neg (by omega), if_pos ⟨h, rfl⟩] theorem Lα_eq : E.Lα = 5 * E.d + 8 + (if E.odd then 2 else 0) := rfl theorem act_last : E.act w (E.Lα - 1) = .send E.r1 := by rw [act, if_neg (by unfold Lα; split <;> omega), if_neg (by unfold Lα; split <;> omega), if_neg (by unfold Lα; split <;> omega), if_neg (by unfold Lα; split <;> omega), if_neg (by unfold Lα; split <;> simp_all), if_neg (by unfold Lα; split <;> simp_all)] @[simp] theorem act_rep (p : ℕ) : (E.act w p).rep = E.r1 := by unfold act; split_ifs <;> rfl theorem act_ins_iff {q : ℕ} {r : Rep} {x : U} {k : ℕ} : E.act w q = .doOp r (.ins x k) ↔ r = E.r1 ∧ ((∃ t ≤ E.d, (q = 3 * t ∧ x = E.Op t ∧ k = E.lenA w t) ∨ (q = 3 * t + 1 ∧ x = E.Cl t ∧ k = E.lenA w t + 1)) ∨ (q = 3 * E.d + 3 ∧ x = E.Star ∧ k = E.lenA w E.d + 1) ∨ (E.odd = true ∧ q = 5 * E.d + 7 ∧ x = E.Bee ∧ k = 0)) := by constructor · intro h unfold act at h split_ifs at h with h1 h2 h3 h4 h5 h6 h7 h8 h9 <;> simp only [Act.doOp.injEq, LOp.ins.injEq, reduceCtorEq, false_and, and_false] at h · obtain ⟨rfl, rfl, rfl⟩ := h exact ⟨rfl, Or.inl ⟨q / 3, by omega, Or.inl ⟨by omega, rfl, rfl⟩⟩⟩ · obtain ⟨rfl, rfl, rfl⟩ := h exact ⟨rfl, Or.inl ⟨q / 3, by omega, Or.inr ⟨by omega, rfl, rfl⟩⟩⟩ · obtain ⟨rfl, rfl, rfl⟩ := h exact ⟨rfl, Or.inr (Or.inl ⟨h4, rfl, rfl⟩)⟩ · obtain ⟨rfl, rfl, rfl⟩ := h exact ⟨rfl, Or.inr (Or.inr ⟨h8.1, h8.2, rfl, rfl⟩)⟩ · rintro ⟨rfl, ⟨t, ht, ⟨rfl, rfl, rfl⟩ | ⟨rfl, rfl, rfl⟩⟩ | ⟨rfl, rfl, rfl⟩ | ⟨h, rfl, rfl, rfl⟩⟩ · exact E.act_Op w ht · exact E.act_Cl w ht · exact E.act_Star w · exact E.act_Bee w h theorem act_del_iff {q : ℕ} {r : Rep} {x : U} : E.act w q = .doOp r (.del x) ↔ r = E.r1 ∧ ((∃ t ≤ E.d, (q = 3 * E.d + 5 + 2 * t ∧ x = E.Op t) ∨ (q = 3 * E.d + 6 + 2 * t ∧ x = E.Cl t)) ∨ (E.odd = true ∧ q = 5 * E.d + 8 ∧ x = E.Bee)) := by constructor · intro h unfold act at h split_ifs at h with h1 h2 h3 h4 h5 h6 h7 h8 h9 <;> simp only [Act.doOp.injEq, LOp.del.injEq, reduceCtorEq, false_and, and_false] at h · obtain ⟨rfl, rfl⟩ := h exact ⟨rfl, Or.inl ⟨(q -
