import Podc.Computability.Distributed.AnonymousNetworks.Lifting /-! # Counterexamples to the impossibility half of Theorem 4.1 and Corollary 4.2 In the paper's model every agent receives its own message on its self-loop, so in round 1 an agent learns its indegree. In a strongly connected network the indegree is `1` iff the agent is alone. Hence the function "`1` if the network has a single agent, `0` otherwise" (`isSingleton`) is computable — in every communication model, even simple broadcast — although it is not frequency-based: `isSingleton [a] = 1 ≠ 0 = isSingleton [a, a]`. With symmetric communications an agent also learns, in round 2, the indegrees of its neighbours: both endpoints of an edge have degree one (one neighbour besides themselves) iff the network is `K₂`. Hence `isPair` ("the network has exactly two agents") is computable with symmetric communications, and is not frequency-based on vectors of length `≥ 2`: `isPair [a, b] = 1 ≠ 0 = isPair [a, b, a, b]`. Both algorithms are self-stabilizing (the output is correct from round 1, resp. 2, whatever the initial configuration), so the counterexamples do not depend on how initial states are chosen. -/ namespace Podc.KnowYourAudience open Cslib.AnonNet Finset Filter Topology universe u variable {I : Type u} /-- "The network has a single agent." -/ def isSingleton (l : List I) : ℕ := if l.length = 1 then 1 else 0 /-- "The network has exactly two agents." -/ def isPair (l : List I) : ℕ := if l.length = 2 then 1 else 0 /-- Count the received messages; the output is `1` iff exactly one message was received. -/ def detectOne (I : Type u) : Algorithm I ℕ where Q := I × Bool M := PUnit init a := (a, false) send _ _ _ := PUnit.unit trans q R := (q.1, decide (Multiset.card R = 1)) out q := if q.2 then 1 else 0 lemma detectOne_allowed (m : CommModel) : m.Allowed (detectOne I) := by cases m <;> simp [CommModel.Allowed, detectOne] lemma detectOne_out {n : ℕ} (N : Network (Fin n)) (hsc : N.StronglyConnected) (C₀ : Fin n → (detectOne I).Q) (t : ℕ) (i : Fin n) : (detectOne I).out (N.runFrom (detectOne I) C₀ (t + 1) i) = if n = 1 then 1 else 0 := by rw [Network.runFrom_succ] simp only [Network.step, Network.recv] have := N.card_inN_eq_one_iff hsc i rw [Fintype.card_fin] at this by_cases h : n = 1 <;> simp [detectOne, h, this] /-- `isSingleton` is computable in the class of static strongly connected networks, in every communication model (from any initial configuration, all outputs are correct from round 1 on). -/ theorem isSingleton_computable (m : CommModel) : Computable m staticSC (fun _ => True) (isSingleton (I := I)) := by refine ⟨detectOne I, detectOne_allowed m, fun n _ N hN _ x _ i => ?_⟩ rw [tendsto_atTop_nhds] intro U hU _ refine ⟨1, fun t ht => ?_⟩ obtain ⟨t, rfl⟩ : ∃ s, t = s + 1 := ⟨t - 1, by omega⟩ rw [Network.run, detectOne_out N hN] simpa [isSingleton] using hU lemma isSingleton_not_frequencyBased [Nonempty I] [DecidableEq I] : ¬ FrequencyBased (isSingleton (I := I)) := by intro h obtain ⟨a⟩ := ‹Nonempty I› have := h [a] [a, a] (by simp) (by simp) (by rw [freq_eq_freq_iff (by simp) (by simp)]; Later rounds: broadcast it; the output is `1` iff the agent and all its in-neighbours have indegree `2`. -/ def detectTwo (I : Type u) : Algorithm I ℕ where Q := I × ℕ × Bool M := ULift ℕ init a := (a, 0, false) send q _ _ := ⟨q.2.1⟩ trans q R := (q.1, Multiset.card R, decide (Multiset.card R = 2 ∧ ∀ r ∈ R, r.down = 2)) out q := if q.2.2 then 1 else 0 lemma detectTwo_allowed : CommModel.symmetric.Allowed (detectTwo I) := by simp [CommModel.Allowed, detectTwo] lemma detectTwo_deg {n : ℕ} (N : Network (Fin n)) (C₀ : Fin n → (detectTwo I).Q) (t : ℕ) (i : Fin n) : (N.runFrom (detectTwo I) C₀ (t + 1) i).2.1 = (N.inN i).card := by rw [Network.runFrom_succ] simp [Network.step, Network.recv, detectTwo] /-- In a bidirectional strongly connected network, an agent of degree `2` whose in-neighbours all have degree `2` lives in a network with two agents, and conversely. -/ lemma pair_iff {n : ℕ} {N : Network (Fin n)} (hsc : N.StronglyConnected) (hsym : N.Symmetric) (i : Fin n) : ((N.inN i).card = 2 ∧ ∀ j ∈ N.inN i, (N.inN j).card = 2) ↔ n = 2 := by constructor · rintro ⟨hi, hj⟩ obtain ⟨a, b, hab, hiab⟩ := Finset.card_eq_two.1 hi -- `j` is the other vertex have hii := N.self_mem_inN i obtain ⟨j, hjmem, hji⟩ : ∃ j ∈ N.inN i, j ≠ i := by rw [hiab] at hii ⊢ rcases Finset.mem_insert.1 hii with h | h · exact ⟨b, by simp, fun h' => hab (h ▸ h'.symm ▸ rfl)⟩ · rw [Finset.mem_singleton] at h exact ⟨a, by simp, fun h' => hab (h ▸ h' ▸ rfl)⟩ have hinI : N.inN i = {i, j} := by symm;
