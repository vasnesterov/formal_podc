import Papers.PODC2024.HistoryIndependent.Alg4 /-! # Algorithm 4: abstract state with ghost variables `St K` is the state of Algorithm 4 (memory, writer's and reader's local state) together with *ghost* variables, and `sstep` its (total) transition function on labels. The ghost state of an execution `α` is `sOf v₀ α = α.foldl sstep init`, a function of the execution. The simulation lemma `rel_run` shows that the concrete part of `sOf v₀ α` is the configuration reached by `α`. Ghost variables (writes are numbered `1, 2, …` in invocation order, `0` is the initial value): * `wv` — the values of the invoked writes; `n = wv.length`; * `tagA j` / `tagB j` — number of the write that last wrote `1` to `A[j]` / `B[j]`; * `sN1` — `n` when the current `Read` executed Line 1; `sT` — `nA` (number of writes that executed Line 16) when the current `TryRead` started; * `lo` — number of completed writes when the current `Read` was invoked; * `rf` — the *reads-from* candidate of the current `Read`: when the reader sets `val ← j` after reading `A[j] = 1` it records `tagA j`; after reading `B[j] = 1` it records `tagB j - 1` (the write *preceding* the one that wrote `B[j]`, whose value is `last-val`); * `fl` — reads-from value of the last completed `Read`; `pis` — reads-from values of all completed `Read`s. -/ set_option linter.unusedSectionVars false namespace Cslib.SharedMemory namespace Alg4 variable {K : ℕ} [NeZero K] /-- Abstract state with ghost variables. -/ structure St (K : ℕ) where A : Fin K → Bool Bm : Fin K → Bool f1 : Bool f2 : Bool last : Fin K wpc : WPC K rpc : RPC K wv : List (Fin K) tagA : Fin K → ℕ tagB : Fin K → ℕ sN1 : ℕ sT : ℕ lo : ℕ rf : ℕ fl : ℕ pis : List ℕ /-- The writer is before Line 16 of a `Write`. -/ def WPC.preA : WPC K → Bool | .scanB .. | .rdF1 _ | .setB _ | .rdF2 _ | .rdF1' _ | .clrB _ | .setA _ => true | _ => false /-- The writer is after Line 16 of a `Write`. -/ def WPC.postA : WPC K → Bool | .down .. | .up .. | .done _ => true | _ => false namespace St /-- Number of invoked writes. -/ def n (s : St K) : ℕ := s.wv.length /-- Number of writes that executed Line 16. -/ def nA (s : St K) : ℕ := if s.wpc.preA then s.n - 1 else s.n /-- Number of completed writes. -/ def c (s : St K) : ℕ := if s.wpc = .idle then s.n else s.n - 1 /-- Value of write number `m` (`0`: initial value). -/ def vals (v₀ : Fin K) (s : St K) (m : ℕ) : Fin K := (v₀ :: s.wv).getD m v₀ end St /-- Initial abstract state. -/ def St.init (v₀ : Fin K) : St K where A j := decide (j = v₀) Bm _ := false f1 := false f2 := false last := v₀ wpc := .idle rpc := .idle wv := [] tagA _ := 0 tagB _ := 0 sN1 := 0 sT := 0 lo := 0 rf := 0 fl := 0 pis := [] /-- Writer's internal step. -/ def stepW (s : St K) : St K := match s.wpc with | .scanB v j => { s with wpc := WPC.scanNext v j (s.Bm j) } | .rdF1 v => { s with wpc := if s.f1 then .setB v else .setA v } | .setB v => { s with Bm := Function.update s.Bm s.last true, tagB := Function.update s.tagB s.last s.n, wpc := .rdF2 v } | .rdF2 v => { s with wpc := if s.f2 then .clrB v else .rdF1' v } | .rdF1' v => { s with wpc := if s.f1 then .setA v else .clrB v } | .clrB v => { s with Bm := Function.update s.Bm s.last false, wpc := .setA v } | .setA v => { s with A := Function.update s.A v true, tagA := Function.update s.tagA v s.n, wpc := WPC.afterSetA v } | .down v j => { s with A := Function.update s.A j false, wpc := WPC.downNext v j } | .up v j => { s with A := Function.update s.A j false, wpc := WPC.upNext v j } | _ => s /-- Reader's internal step. -/ def stepR (s : St K) : St K := match s.rpc with | .start => { s with f1 := true, rpc := .up false 0, sN1 := s.n, sT := s.nA } | .up snd j => { s with rpc := if s.A j then RPC.found j else RPC.upNext snd j, rf := if s.A j then s.tagA j else s.rf, sT := if s.A j = false ∧ snd = false ∧ ¬ (j.val + 1 < K) then s.nA else s.sT } | .down val j => { s with rpc := RPC.downNext val j (s.A j), rf := if s.A j then s.tagA j else s.rf } | .bscan j val => { s with rpc := RPC.bscanNext j val (s.Bm j), rf := if s.Bm j then s.tagB j - 1 else s.rf } | .setF2 val => { s with f2 := true, rpc := .clr 0 val } | .clr j val => { s with Bm := Function.update s.Bm j false, rpc := RPC.clrNext j val } | .resF1 val => { s with f1 := false, rpc := .resF2 val } | .resF2 val => { s with f2 := false, rpc := .done val } | _ => s variable (v₀ : Fin K) /-- The abstract transition function. -/ def sstep (s : St K) : Lbl v₀ → St K | .inv .w (.write v) => { s with wpc := .scanB v 0, wv := s.wv ++ [v] } | .inv .r .read => { s with rpc := .start, lo := s.c } | .inv _ _ => s | .tau .w => stepW s | .tau .r => stepR s | .ret .w _ => match s.wpc with | .done v => { s with last := v, wpc := .idle } | _ => s | .ret .r _ => match s.rpc with | .done _ => { s with rpc := .idle, pis := s.pis ++ [s.rf], fl := s.rf } | _ => s /-- The abstract state (with ghost variables) at the end of an execution. -/ def sOf (α : List (Lbl v₀)) : St K := α.foldl (sstep v₀) (St.init v₀) /-- Which labels are enabled in an abstract state. -/ def Enabled (s : St K) : Lbl
