Every vertex of round `1` has strong edges to `G0`, every vertex of round `r ≥ 2` to all four vertices of round `r - 1`, no weak edges, and block `r`. We compute the states in closed form (`st_eq`: after `t + 1` steps a process is in round `t + 1` and its DAG consists of all vertices of rounds `1..t`), show `R.Valid 1 G0 blk` (`valid`), and that every wave is "good": every vertex of round `wround w 4` has a strong path to every vertex of round `wround w 1` (`good_waves`). -/ namespace DagRider.NonVacuity open Finset /-- Genesis sources. -/ def G0 : Finset (Fin 4) := {0, 1, 2} /-- Blocks: `blk k r = r`. -/ def blk : Fin 4 → ℕ → ℕ := fun _ r => r /-- Global contents. -/ def C : Content (Fin 4) ℕ where strong v := if v.1 = 1 then G0 else univ weak _ := ∅ block v := v.1 /-- The lock-step run. -/ def R : Run (Fin 4) ℕ where C := C byz := ∅ dl _ t := if t = 0 then ∅ else univ.image (fun k => (t, k)) /-- All vertices of rounds `1..t`. -/ def upto (t : ℕ) : Finset (Vtx (Fin 4)) := Icc 1 t ×ˢ univ lemma mem_upto {t : ℕ} {v : Vtx (Fin 4)} : v ∈ upto t ↔ 1 ≤ v.1 ∧ v.1 ≤ t := by simp [upto] lemma mem_dl {p : Fin 4} {t : ℕ} {v : Vtx (Fin 4)} : v ∈ R.dl p t ↔ t ≠ 0 ∧ v.1 = t := by rcases v with ⟨r, k⟩ by_cases ht : t = 0 · simp [R, ht] · simp [R, ht, eq_comm] lemma card_strong (v : Vtx (Fin 4)) : 3 ≤ (C.strong v).card := by simp only [C] split_ifs <;> decide lemma predsIn_upto (t : ℕ) (k : Fin 4) : PredsIn G0 C (upto t) (t + 1, k) := by refine ⟨fun s hs => ?_, fun u hu => by simp [C] at hu⟩ by_cases ht : t = 0 · subst ht simpa [C] using hs · simp only [Nat.add_eq_right, ht, ite_false, Nat.add_sub_cancel, mem_upto] omega lemma layer_upto (t : ℕ) (ht : 1 ≤ t) : layer (upto t) t = {t} ×ˢ univ := by ext ⟨r, k⟩ simp only [mem_layer, mem_upto, mem_product, mem_singleton, mem_univ, and_true] omega lemma lstate_eq {s : LState (Fin 4)} {a : ℕ} {D E : Finset (Vtx (Fin 4))} (h1 : s.rnd = a) (h2 : s.dag = D) (h3 : s.buf = E) : s = ⟨a, D, E⟩ := by cases s;
