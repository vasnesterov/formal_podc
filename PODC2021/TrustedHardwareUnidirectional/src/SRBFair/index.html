A.Init F cl (A.initConfig F cl) := by refine ⟨fun p hp => ⟨rfl, by simp [initConfig, hp]⟩, fun p hp => by simp [initConfig, hp], fun _ _ => rfl⟩ lemma card_pos : 0 < Fintype.card (P × P) := Fintype.card_pos /-- The pair of processes scheduled at time `t` (round robin). -/ def pairAt (t : ℕ) : P × P := (Fintype.equivFin (P × P)).symm ⟨t % Fintype.card (P × P), Nat.mod_lt _ card_pos⟩ /-- Process `qs.1` delivers the next message of `qs.2` if possible. -/ def stepPair (c : SRBConfig P M S) (qs : P × P) : SRBConfig P M S := if h : qs.1 ∉ F ∧ c.dl qs.1 qs.2 < (c.log qs.2).length then A.deliver c cl qs.1 qs.2 ((c.log qs.2)[c.dl qs.1 qs.2]) else c /-- One round-robin step: the scheduled pair delivers if possible. -/ def rrStep (c : SRBConfig P M S) (t : ℕ) : SRBConfig P M S := A.stepPair F cl c (pairAt t) /-- The round-robin execution. -/ def rrExec : ℕ → SRBConfig P M S | 0 => A.initConfig F cl | t + 1 => A.rrStep F cl (rrExec t) t lemma rrExec_isExec : (A.system F cl).IsExec (A.rrExec F cl) := by refine ⟨A.initConfig_init F cl, fun t => ?_⟩ show A.Step F cl _ (A.stepPair F cl (A.rrExec F cl t) (pairAt (P := P) t)) ∨ A.stepPair F cl (A.rrExec F cl t) (pairAt (P := P) t) = _ generalize pairAt (P := P) t = qs unfold stepPair split_ifs with h · left; exact Step.deliver _ _ _ _ h.1 (List.getElem?_eq_getElem h.2) · right; rfl /-- **Round-robin executions are fair**: fairness is not vacuous. -/ theorem rrExec_fair : Fair F (A.rrExec F cl) := by intro q hq s k ⟨t0, ht0⟩ _ have he := A.rrExec_isExec F cl set N := Fintype.card (P × P) have hN : 0 < N := card_pos set i := ((Fintype.equivFin (P × P)) (q, s)).val with hi have hiN : i < N := ((Fintype.equivFin (P × P)) (q, s)).isLt have hpair : ∀ τ, τ % N = i → pairAt τ = (q, s) := by intro τ hτ simp only [pairAt] rw [Equiv.symm_apply_eq] apply Fin.ext exact hτ -- the counter grows at every scheduled time while it is ≤ k have key : ∀ j, min j (k + 1) ≤ (A.rrExec F cl (t0 + j * N)).dl q s := by (A.rrExec F cl τ) (pairAt τ)).dl q s = _ rw [hpair τ hτN] unfold stepPair rw [dite_eq_left ⟨hq, by dsimp only; omega⟩] exact deliver_dl_self omega refine ⟨t0 + (k + 1) * N, ?_⟩ have := key (k + 1)
